/*  FUNCTION NAME:  IsOccurrenceDateλ
    DESCRIPTION:*//**Returns TRUE if Date passed is when something happens. This was developed for budgeting where we have expenses that start, and sometimes repeat at regular intervals.*/

IsOccurrenceDateλ = LAMBDA(
//  Parameter declarations
    [Dates],
    [FirstOccurrence],
    [LastOccurrence],
    [Repeats],
//  Help
    LET(
        Help,       TRIM(TEXTSPLIT(
                        "FUNCTION:      →IsOccurrenceDateλ(Dates, FirstOccurrence, [LastOccurrence], [Repeats])¶" &
                        "DESCRIPTION:   →Returns TRUE if a Date is when something happens.¶" &
                        "               →This was developed for budgeting where we have expenses that start,¶" &
                        "               →and sometimes repeat at regular intervals.¶" &  
                        "WEBSITE:       →https://github.com/ryanduguid/Ozzit¶" &
                        "PARAMETERS:    →¶" &
                        "Dates          →(Required) Date to test. Normally this is horizontal timeline by day¶" &
                        "FirstOccurrence →(Required) When something that may repeat, happens for the first time.¶" &
                        "LastOccurrence  →(Optional) When something that may repeat, happens for the last time.¶" &
                                        "→If omitted, it is assumed repeating items repeat forever.¶" &
                        "Repeats        →(Optional) A code that conveys how often this thing repeats. Repeat codes are¶" &
                                        "→O=One time/never repeats (default)¶" & 
                                        "→A=Annually¶" & 
                                        "→S=Semi-Annually¶" &
                                        "→Q=Quarterly¶" & 
                                        "→M=Monthly¶" & 
                                        "→B=BiWeekly¶" & 
                                        "→W=Weekly¶" &
                                        "→X=Every Workday (Monday to Friday)¶" & 
                                        "→D=Daily¶" &                      
                        "NOTE!          →A monthly, quarterly, semi-annual or annual item that starts on the 29th, 30th or 31st¶" &
                                        "→falls on the last day of any shorter month.¶" &
                        "EXAMPLES:      →¶" &
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                        "72             →=SUM(N(oz.IsOccurrenceDateλ(SEQUENCE(,90,""1/7/2026""), ""1/7/2026"", ""10/9/2026"", ""D"")))¶" &
                        "01             →=SUM(N(oz.IsOccurrenceDateλ(SEQUENCE(,90,""1/7/2026""), ""1/7/2026"", ""10/9/2026"", ""O"")))¶" &
                        "11             →=SUM(N(oz.IsOccurrenceDateλ(SEQUENCE(,90,""1/7/2026""), ""1/7/2026"", ""10/9/2026"", ""W"")))", 
                        "→", "¶"
                        )
                    ),
    //  Named constants
        SemiAnnually,   6,    //Months
        Quarterly,      3,    //Months
        Biweekly,       14,   //Days
        Weekly,         7,    //Days
        Mon2Sun,        2,    //WEEKDAY() Type parameter
        Friday,         5,    //WEEKDAY()'s value for Friday when type= 2
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Dates), 
                            ISOMITTED( FirstOccurrence)
                        ),
    //  Check inputs - Set optional arguments' defaults and apply convenience conversions
        CvtDates,       IF( ISOMITTED( Dates), 
                            FALSE,
                            IF( ISNUMBER( Dates), Dates, DATEVALUE( Dates))
                        ),
        CvtFirsts,      IF( ISOMITTED( FirstOccurrence), 
                            FALSE, 
                            IF( ISNUMBER( FirstOccurrence), FirstOccurrence, DATEVALUE( FirstOccurrence))
                        ),
    //  Row by row, so a blank or a date in one row of a table column does not decide how
    //  every row is read. A blank last occurrence becomes "", which Test reads as none.
        CvtLasts,       IF( ISOMITTED( LastOccurrence), 
                            FALSE, 
                            IF( TRIM( LastOccurrence) = "", 
                                "", 
                                IF( ISNUMBER( LastOccurrence), LastOccurrence, DATEVALUE( LastOccurrence)))
                        ),
        Repeats,        IF( ISOMITTED( Repeats), "O", IF( TRIM( Repeats) = "", "O", LEFT( Repeats, 1))),
    //  Procedure
    //  A start on the 29th, 30th or 31st falls on the last day of any shorter month
        MonthEnds,      DAY( EOMONTH( CvtDates, 0)),
        SameDay,        DAY( CvtDates) = IF( DAY( CvtFirsts) > MonthEnds, MonthEnds, DAY( CvtFirsts)),
        SameMonth,      MONTH( CvtDates) = MONTH( CvtFirsts),
        Test,           ( CvtFirsts <= CvtDates) * 
                        IF(TRIM( CvtLasts) = "", 1, CvtDates <= CvtLasts) *
                        SWITCH(
                            Repeats,
                            "O",    CvtDates = CvtFirsts,
                            "A",    SameDay * SameMonth,
                            "S",    SameDay * 
                                        (MOD( MONTH( CvtDates), SemiAnnually) = MOD( MONTH( CvtFirsts), SemiAnnually)),
                            "Q",    SameDay *
                                        (MOD( MONTH( CvtDates), Quarterly) = MOD( MONTH( CvtFirsts), Quarterly)),
                            "M",    SameDay,
                            "B",    MOD( CvtDates, Biweekly) = MOD( CvtFirsts, Biweekly),
                            "W",    MOD( CvtDates, Weekly) = MOD( CvtFirsts, Weekly),
                            "X",    WEEKDAY( CvtDates, Mon2Sun) <= Friday,
                            "D",    TRUE,
                            0
                        ),
        Result,         Test = 1,        
    //  Return Result
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  Periodsλ
    DESCRIPTION:*//**Determine the number of periods from Starts to Date2 in Days, Weeks, Months, Quarters, or Years.*/
Periodsλ = LAMBDA(
//  Parameter Declarations
    [Starts],
    [Ends],
    [Intervals],
    LET(
//  Help
        Help,       TRIM(TEXTSPLIT( 
                        "FUNCTION:      →Periodsλ(Starts, Ends, [Intervals])¶" &
                        "DESCRIPTION:   →Determine the number of periods from Starts to Ends inclusive.¶" &
                                       "→Differences to Excel's DATEDIF are:¶" & 
                                       "→* Interval includes Weeks & Quarters¶" & 
                                       "→* Counts the period starts crossed, so a part period at the end counts¶" & 
                                       "→* Returns negative values if Starts is after Ends¶" & 
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                        "PARAMETERS:    →¶" &
                        "Starts         →(Required) First date (normally older). A single date or an array of dates.¶" &
                        "Ends           →(Required) Second date (normally newer). A single date or an array the same shape as Starts.¶" &
                                       "→A single Starts or Ends is applied to every date in the other.¶" &
                        "Intervals      →(Optional) Days, Weeks, Months (default), Quarters, or Years (D, W, M, Q, Y).¶" &
                                       "→One code, or an array of codes the same shape as the dates.¶" &
                        "EXAMPLES:      →¶" & 
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                        "2              →=oz.Periodsλ(""31/7/2026"", ""15/9/2026"")¶" &
                        "-2             →=oz.Periodsλ(""15/9/2026"", ""31/7/2026"")¶" &
                        "-12            →=oz.Periodsλ(""15/7/2027"", ""16/7/2026"")¶" &
                        "-52            →=oz.Periodsλ(""15/7/2027"", ""16/7/2026"", ""W"")",
                        "→", "¶"
                    )
                ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Starts),
                            ISOMITTED( Ends)
                        ),
    //  Check inputs - Set optional arguments' defaults and apply convenience conversions
        CvtStarts,      IF( ISOMITTED( Starts), 
                            FALSE,
                            IF( ISNUMBER( Starts), Starts, DATEVALUE( Starts))
                        ),
        CvtEnds,       IF( ISOMITTED( Ends), 
                            FALSE,
                            IF( ISNUMBER( Ends), Ends, DATEVALUE( Ends))
                        ),
        Intervals,      IF( ISOMITTED( Intervals), "M", Intervals),
    //  Constant Declarations
        MPY, 12,    //Months Per Year
        MPQ, 3,     //Months Per Quarter
        DPW, 7,     //Days Per Week
        WPY, 53,    //Week labels Per Year: the last one runs 1 or 2 days
        QPY, 4,     //Quarters Per Year
    //  Procedure - every step lifts over arrays, so a single date or code applies to all
        Earliest,   IF( CvtStarts <= CvtEnds, CvtStarts, CvtEnds),
        Latest,     IF( CvtStarts <= CvtEnds, CvtEnds, CvtStarts),
        Sign,       SIGN( CvtEnds - CvtStarts),
        Years,      YEAR( Latest) - YEAR( Earliest),
        Periods,    ( Intervals = "D") * ( Latest - Earliest)
                  + ( Intervals = "W") * ( WPY * Years
                                           + QUOTIENT( Latest   - DATE( YEAR( Latest),   1, 0) - 1, DPW)
                                           - QUOTIENT( Earliest - DATE( YEAR( Earliest), 1, 0) - 1, DPW))
                  + ( Intervals = "M") * ( MPY * Years + MONTH( Latest) - MONTH( Earliest))
                  + ( Intervals = "Q") * ( QPY * Years
                                           + QUOTIENT( MONTH( Latest)   - 1, MPQ)
                                           - QUOTIENT( MONTH( Earliest) - 1, MPQ))
                  + ( Intervals = "Y") * Years,
    //  An unknown interval code is an error, not a nought
        Known,      IF( ISNUMBER( XMATCH( Intervals, {"D","W","M","Q","Y"}, 0)), 1, #N/A),
        Result,     Sign * Periods * Known,
    //  Return Result
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  ScheduleRatesByItemsλ

ScheduleRatesByItemsλ = LAMBDA(
//  Parameter Declarations
    [PeriodEnds],
    [ItemFilter],
    [ItemList],
    [EffectiveList],
    [RateList],
//  Help
    LET(
        Help,       TRIM(TEXTSPLIT(
                        "FUNCTION:      →ScheduleRatesByItemsλ(PeriodEnds, ItemFilter, ItemList, EffectiveList, RateList)¶" &
                        "DESCRIPTION:   →Schedules each item's rates by period until another rate takes effect.¶" &

                        "PeriodEnds     →(Required) A row of period end dates (timeline).¶" &

                                       "→An item with no rate in the schedule scores nought in every period.¶" &

                                       "→tblRates[Unit], tblRates[Effective], tblRates[Rent])", 
                        "→", "¶"
                            )
                        ),
                    
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( PeriodEnds), 
                            ISOMITTED( ItemFilter),
                            ISOMITTED( ItemList), 
                            ISOMITTED( EffectiveList),

                        ),

                            FALSE,
                            IF( ISNUMBER( PeriodEnds), PeriodEnds, DATEVALUE( PeriodEnds))
                        ),
        CvtEffDates,    IF( ISOMITTED( EffectiveList), 
                            FALSE,
                            IF( ISNUMBER( EffectiveList), EffectiveList, DATEVALUE( EffectiveList))
                        ),
    //  Procedure - one row per item, each the latest rate in force at every period end.
    //  An item with no rows in the schedule keeps a placeholder date no period can reach,
    //  so it scores nought rather than #CALC!.
        Items,          TOCOL( ItemFilter),
        Result,         DROP( 
                            REDUCE( 0, SEQUENCE( ROWS( Items)),
                                LAMBDA( Acc, Row,
                                    LET(Item,       INDEX( Items, Row),
                                        ItemDates,  FILTER( CvtEffDates, ItemList = Item, -1), 
                                        ItemRates,  FILTER( RateList, ItemList = Item, 0), 
                                        VSTACK( Acc, XLOOKUP( CvtPrdEnds, ItemDates, ItemRates, 0, -1, -1))
                                    )
                                )
                            ),
                            1
                        ),
    //  Return Result
        CHOOSE( Help? + 1, Result, Help)
    )
);

    [PeriodStarts],

    [Values],
    [EffectiveDates],
    LET(

        Help,       TRIM(TEXTSPLIT(

                        "DESCRIPTION:   →Place values from a table into a timeline's appropriate period.¶" &

                        "Values         →(Required) A column or row of values to be scheduled.¶" & 
                        "EffectiveDates →(Required) When each value is scheduled, in the same orientation as Values.¶" &                   
                        "EXAMPLES:→¶" &
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                                       "→=oz.ScheduleValuesλ(Timelineλ(""1/7/2026""), Timelineλ(""1/7/2026"",,,FALSE), {1;3}, {""1/7/2026""; ""15/8/2026""})", 
                        "→", "¶"
                        )
                    ),

        Help?,          OR( ISOMITTED( PeriodStarts), 
                            ISOMITTED( PeriodEnds),
                            ISOMITTED( Values),

                        ),

        CvtPrdStarts,   IF( ISOMITTED( PeriodStarts),

                            IF( ISNUMBER( PeriodStarts), PeriodStarts, DATEVALUE( PeriodStarts))

                            IF( ISNUMBER( PeriodEnds), PeriodEnds, DATEVALUE( PeriodEnds))
                        ),

                            FALSE,
                            IF( ISNUMBER( EffectiveDates), EffectiveDates, DATEVALUE( EffectiveDates))
                        ),

        Result,         BYCOL(
                            SEQUENCE(, COLUMNS(CvtPrdStarts)), 
                            LAMBDA(Period, 
                                LET(StartDate,  INDEX(CvtPrdStarts, Period), 
                                    EndDate,    INDEX(CvtPrdEnds, Period),
                                    SUM(Values * (CvtEffDates >= StartDate) * (CvtEffDates <= EndDate))
                                )
                            )
                        ),

        CHOOSE( Help? + 1, Result, Help)
    )

/*  FUNCTION NAME:  ScheduleValuesByItemsλ

ScheduleValuesByItemsλ = LAMBDA(
//  Parameter Declarations
    [PeriodStarts],
    [PeriodEnds],
    [ItemFilter],
    [ItemList],
    [EffectiveList],
    [ValueList],
    LET(
//  Help
        Help,       TRIM(TEXTSPLIT( 
                        "FUNCTION:      →ScheduleValuesByItemsλ(PeriodStarts, PeriodEnds, ItemFilter, ItemList, EffectiveList, ValueList)¶" &
                        "DESCRIPTION:   →Schedule and sum values in a timeline for each item.¶" & 
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                        "PARAMETERS:    →¶" & 
                        "PeriodStarts   →(Required) A row of period start dates (timeline).¶" & 
                        "PeriodEnds     →(Required) A row of period end dates (timeline).¶" & 
                        "ItemFilter     →(Required) A column of 1 or more items to filter a value schedule by. ¶" &
                        "               →A 'value schedule' is a table/list where each row contains a unique item/effective date pair with a value.¶" &
                        "ItemList       →(Required) Items in a value schedule.¶" & 
                        "EffectiveList  →(Required) Effective dates in the value schedule.¶" & 
                        "ValueList      →(Required) Values in a value schedule for each item/effective date pair.¶" & 
                                       "→An item with no rows in the schedule totals nought in every period.¶" & 
                        "EXAMPLES:      →¶" & 
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                                       "→=oz.ScheduleValuesByItemsλ( oz.Timelineλ( ""1/7/2026""),  oz.Timelineλ( ""1/7/2026"",,,FALSE), ¶" & 
                                       "→UNIQUE( tblExp[Items]), tblExp[Items], tblExp[Dates], tblExp[Amount])", 
                        "→", "¶"
                        )
                    ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( PeriodStarts), 
                            ISOMITTED( PeriodEnds), 
                            ISOMITTED( ItemFilter),
                            ISOMITTED( ItemList), 
                            ISOMITTED( EffectiveList),
                            ISOMITTED( ValueList)
                        ),
    //  Check inputs - Set optional arguments' defaults and apply convenience conversions
        CvtPrdStarts,   IF( ISOMITTED( PeriodStarts), 
                            FALSE,
                            IF( ISNUMBER( PeriodStarts), PeriodStarts, DATEVALUE( PeriodStarts))
                        ),
        CvtPrdEnds,     IF( ISOMITTED( PeriodEnds), 
                            FALSE,
                            IF( ISNUMBER( PeriodEnds), PeriodEnds, DATEVALUE( PeriodEnds))
                        ),
        CvtEffDates,    IF( ISOMITTED( EffectiveList), 
                            FALSE,
                            IF( ISNUMBER( EffectiveList), EffectiveList, DATEVALUE( EffectiveList))
                        ),   
    //  Procedure - one matrix product: which schedule rows belong to each item, times
    //  which period each schedule row falls in, carrying its value
        Items,          TOCOL( ItemFilter),
        Belongs,        ( Items = TOROW( ItemList)) * ( Items <> ""),
        Falls,          ( TOCOL( CvtEffDates) >= CvtPrdStarts) * ( TOCOL( CvtEffDates) <= CvtPrdEnds) * TOCOL( ValueList),
        Result,         MMULT( Belongs, Falls),
    //  Return Result
        CHOOSE( Help? + 1, Result, Help)
    )
);

    DESCRIPTION:*//**Create a horizontal list of period dates*/
Timelineλ = LAMBDA(
//  Parameter Declarations
    [StartDate],
    [Periods],
    [Interval],
    [PeriodStarts?],    
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:      →Timelineλ(StartDate, Periods, Interval, PeriodStarts?)¶" &
                            "DESCRIPTION:   →Creates a horizontal list of period dates.¶" & 
                            "WEBSITE:       →https://github.com/ryanduguid/Ozzit¶" &

                            "Periods        →(Optional) The number of period in the timeline. Defaults to 12.¶" &
                            "Interval       →(Optional) Period interval: D, W, M, Q or Y (Day, Week, Month <default>, Quarter, or Year)¶" &
                            "PeriodStarts?  →(Optional) TRUE=Display period start dates <default>. FALSE=Period end dates¶" &
                            "EXAMPLES:      →¶" & 
                            "Result         →Formula (oz is assumed to be the module's name)¶" &
                            "12 mo, starts  →=oz.Timelineλ(TODAY())¶" &
                            "12 qtr, ends   →=oz.Timelineλ(TODAY(), , ""Q"", FALSE)",
                            "→", "¶"
                            )
                        ),      
    //  Named Constants
        MPY, 12,        //Months Per Year
        MPQ, 3,         //Months Per Quarter
        DPW, 7,         //Days Per Week
    //  Check inputs - Omitted required arguments
        Help?,          ISOMITTED( StartDate), 
    //  Check inputs - Set optional arguments' defaults and apply convenience conversions
        CvtStart,       IF( ISOMITTED( StartDate), 
                            FALSE,
                            IF( ISNUMBER( StartDate), StartDate, DATEVALUE( StartDate))
                        ),
        Periods,        IF( ISOMITTED( Periods), 12, Periods),
        Interval,       IF( ISOMITTED( Interval), "M", Interval),
        PeriodStarts?,  IF( ISLOGICAL( PeriodStarts?), PeriodStarts?, TRUE),
    //  Procedure
        EndDates,       N(NOT(PeriodStarts?)),
        Result,         SWITCH(
                            Interval,
                            "Y", EDATE(CvtStart, SEQUENCE(1, Periods, EndDates * MPY, MPY)) - EndDates,
                            "Q", EDATE(CvtStart, SEQUENCE(1, Periods, EndDates * MPQ, MPQ)) - EndDates,
                            "M", EDATE(CvtStart, SEQUENCE(1, Periods, EndDates, 1)) - EndDates,
                            "W", SEQUENCE(1, Periods, CvtStart + IF(PeriodStarts?, 0, DPW - 1), DPW),
                            "D", SEQUENCE(1, Periods, CvtStart, 1),
                            #Value!
                        ),
    //  Return Result
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  DateDifλ
    DESCRIPTION:*//**Whole years, months or days between two dates, and the remainders DATEDIF gets wrong*/

DateDifλ = LAMBDA(
//  Parameter Declaration
    [StartDate],
    [EndDate],
    [Unit],
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:      →DateDifλ(StartDate, EndDate, [Unit])¶" &
                            "DESCRIPTION:   →Whole years, months or days between two dates, and the remainders.¶NOTES!         →Excel documents DATEDIF as a legacy compatibility function and warns¶               →its MD unit can return a negative or wrong day count. This counts a¶               →month as complete when EDATE() of the start date has arrived, so a¶               →month after 31 January is the last day of February, and takes every¶               →remainder from there. A reversed range or an unknown unit returns #NUM!.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                            "VERSION:       →6 Sep 2026¶" &
                            "PARAMETERS:    →¶" &
                            "StartDate      →(Required) The earlier date, numeric or text.¶" &
                            "EndDate        →(Required) The later date, numeric or text.¶" &
                            "Unit           →(Optional: Default = D) Y whole years, M whole months, D days, YM months left after the whole years, YD days left after the whole years, MD days left after the whole months.¶" &
                            "EXAMPLES:      →¶" &
                            "→Formula (oz is assumed to be the module's name)¶" &
                            "→=oz.DateDifλ(DATE(2026,1,31), DATE(2026,3,1), ""MD"")¶" &
                            "→Result¶" &
                            "→1",
                            "→", "¶"
                        )),
    //  Check inputs - Omitted required arguments
        Help?,          OR(ISOMITTED(StartDate), ISOMITTED(EndDate)),
    //  Text dates are read; a blank Unit cell is not an omitted argument, so test for both
        Start,          IF(ISNUMBER(StartDate), StartDate, DATEVALUE(StartDate)),
        Finish,         IF(ISNUMBER(EndDate), EndDate, DATEVALUE(EndDate)),
    //  Element-wise, so one blank Unit in a column defaults its own row and not every row
        Code,           IF(ISOMITTED(Unit), "D", IF(Unit = "", "D", UPPER(Unit))),
    //  Calendar months crossed, less one when the last has not yet completed
        RawMonths,      12 * (YEAR(Finish) - YEAR(Start)) + MONTH(Finish) - MONTH(Start),
        Months,         RawMonths - (EDATE(Start, RawMonths) > Finish),
        Years,          INT(Months / 12),
        Result,         IF(Finish < Start,
                            #NUM!,
                            SWITCH(Code,
                                "Y", Years,
                                "M", Months,
                                "D", Finish - Start,
                                "YM", Months - 12 * Years,
                                "YD", Finish - EDATE(Start, 12 * Years),
                                "MD", Finish - EDATE(Start, Months),
                                #NUM!)),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  Corkscrewλ
    DESCRIPTION:*//**Create a corkscrew section that sums the opening balance and up to 4 independent flows.*/
Corkscrewλ = LAMBDA(
//  Parameter Declarations
    [Opening],
    [Flow1],
    [Flow2],
    [Flow3],
    [Flow4],

    LET(
    //  Help
        Help,       TRIM(TEXTSPLIT( 
                        "FUNCTION:      →Corkscrewλ( Opening, Flow1, [Flow2], [Flow3], [Flow4])¶" &
                        "DESCRIPTION:   →Create a corkscrew section that sums the opening balance and up to 4 independent flows¶" & 
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                        "VERSION:       →20 Aug 2026¶" &
                        "PARAMETERS:→¶" & 
                        "Opening        →(Required) First periods opening balance.¶" & 
                        "Flow1          →(Required) First row (or array of rows) of numbers to include in period totals.¶" & 
                        "Flow2          →(Optional) Second row (or array of rows) of numbers to include in period totals.¶" & 
                        "Flow3          →(Optional) Third row (or array of rows) of numbers to include in period totals.¶" & 
                        "Flow4          →(Optional) Fourth row (or array of rows) of numbers to include in period totals.¶" & 
                        "NOTE!          →All flows are added. To subtract a flow make it negative.¶" & 
                        "→¶" &
                        "EXAMPLES:      →NOTE! oz is assumed to be the module's name¶" & 
                        "Formula:       →=oz.Corkscrewλ(0, Sequence(1,5))¶" &
                        "Result:        →00, 01, 03, 06, 10¶" &
                                       "→01, 02, 03, 04, 05¶" &
                                       "→01, 03, 06, 10, 15", 
                        "→", "¶" )
                    ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Opening),
                            ISOMITTED( Flow1)
                        ),
    //  Processing
        Cols,           Columns(Flow1),
        Rows,           IF(ISOMITTED(Flow2), 1, 
                            IF(ISOMITTED(Flow3), 2, 
                                IF(ISOMITTED(Flow4), 3, 4)
                            )
                        ),                        
        PeriodArray,    SWITCH(Rows, 
                            1,  Flow1, 
                            2,  VSTACK(Flow1, Flow2), 
                            3,  VSTACK(Flow1, Flow2, Flow3),
                            4,  VSTACK(Flow1, Flow2, Flow3, Flow4)
                        ), 
        PeriodTotal,    BYCOL(PeriodArray, LAMBDA(Col, SUM(Col))), 
        ClosingTotal,   SCAN(Opening, PeriodTotal, LAMBDA(Opn, PrdTot, Opn + PrdTot)), 
        OpeningTotal,   MAP( SEQUENCE(, Cols), LAMBDA( Col, IF( Col = 1, Opening, INDEX( ClosingTotal, Col - 1)))),
        Result,         VSTACK(OpeningTotal, PeriodArray, ClosingTotal), 
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )  
);



/*  FUNCTION NAME:  CorkScrewReversalλ
    DESCRIPTION:*//**Create a corkscrew where the opening balance is reversed in the next period based on a timing parameter.*/
CorkScrewReversalλ=LAMBDA(
//  Parameter Declarations
    [Opening],
    [ReversalFlags],
    [Flow1],
    [Flow2],
    [Flow3],
    [Flow4], 
    //  Help
    LET(Help,           TRIM(TEXTSPLIT(
                        "FUNCTION:      →CorkScrewReversalλ( Opening, ReversalFlags, Flow1, [Flow2], [Flow3], [Flow4])¶" & 
                        "DESCRIPTION:   →Create a corkscrew where the opening balance is reversed in the next period based on a timing parameter¶" & 
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                        "VERSION:       →20 Aug 2026¶" & 
                        "PARAMETERS:→¶" & 
                        "Opening        →(Required) First periods opening balance.¶" & 
                        "ReversalFlags  →(Required) A row of ones and blanks (or zeros) indicating when to reverse period totals.¶" & 
                        "Flow1          →(Required) First row (or array of rows) of numbers to include in period totals.¶" & 
                        "Flow2          →(Optional) Second row (or array of rows) of numbers to include in period totals.¶" & 
                        "Flow3          →(Optional) Third row (or array of rows) of numbers to include in period totals.¶" & 
                        "Flow4          →(Optional) Fourth row (or array of rows) of numbers to include in period totals.¶" & 
                        "NOTE!          →All flows are added. To subtract a flow make it negative.", 
                        "→", "¶")), 
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Opening), 
                            ISOMITTED( ReversalFlags), 
                            ISOMITTED( Flow1)), 
    //  Set defaults and constants
        Cols,           COLUMNS(Flow1), 
        Rows,           IF(ISOMITTED(Flow2), 1, 
                            IF(ISOMITTED(Flow3), 2, 
                                IF(ISOMITTED(Flow4), 3, 4)
                            )
                        ), 
    //  Procedure
        PeriodArray,    SWITCH(Rows, 
                            1, Flow1, 
                            2, VSTACK(Flow1, Flow2), 
                            3, VSTACK(Flow1, Flow2, Flow3), 
                            4, VSTACK(Flow1, Flow2, Flow3, Flow4)
                        ), 
        PeriodTotal,    BYCOL(PeriodArray, LAMBDA(Col, SUM(Col))), 
    //  The closing balance is the only state carried from one period to the next: a period
    //  closes on its opening balance, less that balance where the flag reverses it, plus
    //  its flows. Solve the closings in one pass and read every other row off them.
        Flags,          INDEX( ReversalFlags, 1, SEQUENCE( , Cols)),
        Closings,       SCAN( Opening, SEQUENCE( , Cols),
                            LAMBDA( Acc, Col,
                                Acc * ( 1 - INDEX( Flags, 1, Col)) + INDEX( PeriodTotal, 1, Col)
                            )
                        ),
        Openings,       IF( Cols = 1, Opening, HSTACK( Opening, DROP( Closings, , -1))),
        Reversals,      -Flags * Openings,
        Result,         VSTACK( Openings, Reversals, PeriodArray, Closings),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )  
);

//  Amortisation Suite

/*  FUNCTION NAME:  Amortiseλ
    DESCRIPTION:*//**Creates a corkscrew amortisation schedule*/
Amortiseλ = LAMBDA(
//  Parameter Declarations
    [Principals], 
    [APRs], 
    [Terms], 
    [StartDates],
    [Timeline],
    [FinalPeriodEnd],

    LET(
    //  Help        
        Help,       TRIM(TEXTSPLIT( 
                        "FUNCTION:      →Amortiseλ( Principals, APRs, Terms, StartDates, [Timeline], [FinalPeriodEnd])¶" &
                        "DESCRIPTION:   →Creates an amortisation schedule array for one or more loans.¶" &
                                       "→Payments are assumed monthly. On periods shorter than a month the figures for each month land in the period holding its start, the rest nil.¶" &
                                       "→Use LabelAmortiseλ() to apply labels to this function's results:¶" & 
                                       "→Use SumAmortiseλ() to apply totals to this function's results:¶" & 
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                        "VERSION:       →20 Aug 2026¶" &
                        "PARAMETERS:→¶" & 
                        "Principals     →(Required) Loan/debt amounts.¶" & 
                        "APRs           →(Required) Annual Percentage Rates for interest.¶" & 
                        "Terms          →(Required) Number of months for loans/debts.¶" & 
                        "StartDates     →(Required) Dates debts are issued.¶" & 
                        "Timeline       →(Optional) An array of dates. If omitted, a row of months starting on start date will be added¶" & 
                        "FinalPeriodEnd →(Optional) The day after the last period ends. Needs Timeline, and only moves money where its periods are shorter than a month. If omitted, the last period runs as long as the one before it.¶" &
                        "EXAMPLES:→¶" &
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                                       "→=oz.Amortiseλ(10000, 5%, 4 * 12, ""2026-07-01"")", 
                        "→", "¶" )
                    ),
    //  Constants
        MpY,                12,     //Months Per Year
        ADpM,               30.42,  //Average Days Per Month
    
    //  Check inputs - Omitted required arguments
        Help?,              OR( ISOMITTED( Principals),
                                ISOMITTED( APRs),
                                ISOMITTED( Terms),
                                ISOMITTED( StartDates)
                            ),
    
    //  Set defaults and apply convenience conversions. A text start date is read as a date,
    //  and the default timeline runs from the earliest start to the last month of the
    //  longest-running loan, whichever loan that is.
        CvtStarts,          IF( ISNUMBER( StartDates), StartDates, DATEVALUE( StartDates)),
    //  Read before the default replaces it, because FinalPeriodEnd describes a supplied
    //  timeline and the default is monthly, where it would do nothing.
        TimelineGiven?,     NOT( ISOMITTED( Timeline)),
        Timeline,           IF( ISOMITTED( Timeline), 
                                LET(LastMonths, EDATE( CvtStarts, Terms - 1),
                                    MinDate,    MIN( CvtStarts),
                                    LastMonth,  MAX( LastMonths),
                                //  Count calendar months, not DATEDIF "M": a month-end start
                                //  lands its last EDATE in a shorter month and DATEDIF drops it.
                                    MaxTerm,    ( YEAR( LastMonth) - YEAR( MinDate)) * MpY
                                                + MONTH( LastMonth) - MONTH( MinDate) + 1,
                                    EDATE( MinDate, SEQUENCE( 1, MaxTerm, 0))),
                                Timeline
                            ),
    //  Determine timeline's interval (length of periods) and orientation (horizontal or vertical)
    //  A one-date timeline (a one-month loan) has no second date to measure, so it is monthly.
        FirstPeriod,        INDEX( Timeline, 1),
        Single?,            COUNTA( Timeline) < 2,
        SecondPeriod,       IF( Single?, EDATE( FirstPeriod, 1), INDEX( Timeline, 2)),
    //  A supplied timeline has to advance. The approximate MATCH below buckets a repeated
    //  or out-of-order date as one period, which moves money into a period the dates do not
    //  justify instead of failing: 1 January to 1 June with 1 March repeated returned
    //  5,772.61 of principal where the same span without the repeat returns 4,937.63. A
    //  one-date timeline has no pair to measure. The default timeline always advances.
        BadTimeline?,       IF( Single?, FALSE,
                                LET( Dates, TOROW( Timeline),
                                     MIN( DROP( Dates, , 1) - DROP( Dates, , -1)) < 1)),
        TimelineMessage,    "Timeline must be period start dates where each date is at least a day "
                            & "after the one before it. A repeated or out-of-order date puts one period's "
                            & "money in another period.",
        DpP,                SecondPeriod - FirstPeriod,                 //Days Per Period
    //  The last period has no successor to measure, so it runs as long as the one before
    //  it. FinalPeriodEnd states that boundary instead: it is the day after the period
    //  ends, so it has to fall after the last period start. It describes a timeline the
    //  caller supplied, so it is refused without one rather than measured against a
    //  default this function chose. COUNT rejects text and a range of more than one cell
    //  without reading an omitted argument.
        FinalEnd?,          NOT( ISOMITTED( FinalPeriodEnd)),
        FinalEndDate,       IF( FinalEnd?, MAX( FinalPeriodEnd), 0),
        BadFinalEnd?,       IF( FinalEnd?,
                                OR( NOT( TimelineGiven?),
                                    COUNT( FinalPeriodEnd) <> 1,
                                    FinalEndDate <= MAX( Timeline)),
                                FALSE),
        FinalEndMessage,    "FinalPeriodEnd needs a Timeline, and must be one date after the last "
                            & "date in it. It is the day after the final period ends.",
        LastGap,            IF( FinalEnd?,
                                FinalEndDate - MAX( Timeline),
                                IF( Single?, DpP, MAX( Timeline) - INDEX( Timeline, COUNTA( Timeline) - 1))),
        SubMonthly?,        DpP < 28,                                   //No month is shorter
        MpP,                MAX( @ROUND( DpP / ADpM, 0), 1),            //Months Per Period, never nought

    //  One loan's schedule, laid onto the timeline
        Schedule, 
            LAMBDA( Asset,
                LET(
                //  Create individual Values from array
                    Principal,          INDEX( Principals, Asset), 
                    APR,                INDEX( APRs, Asset), 
                    Term,               INDEX( Terms, Asset),  
                    StartDate,          INDEX( CvtStarts, Asset), 

                //  Adjust inputs to match timeline
                    Periods,            ROUNDUP(Term / MpP, 0),
                    Rows,               1,
                    Cols,               Periods,

                //  Calculate for each month
                    Rate,               APR / MpY,
                    MonthlyPayment,     PMT( Rate, Term, Principal),    
                    MonthlyPrincipal,   -FV(Rate, SEQUENCE( 1, Term, 0), MonthlyPayment, Principal),
                    MonthlyInterest,    MonthlyPrincipal * Rate,

                //  Convert to Timeline Periods
                    DebtIssueArray,     EXPAND(Principal, 1, Periods, 0),
                    PrincipalArray,     IF( MpP = 1, 
                                            MonthlyPrincipal, 
                                            MAKEARRAY( 1, Periods, 
                                                LAMBDA(R, C, 
                                                    LET(Idx, (C - 1) * MpP + 1, 
                                                        INDEX( MonthlyPrincipal, Idx)
                                                    )
                                                )
                                            )
                                        ),    
                    InterestArray,      IF( MpP = 1, 
                                            MonthlyInterest,
                                            LET( 
                                                MoInt,  EXPAND( MonthlyInterest, 1, ROUNDUP( Term/ Mpp, 0) * Mpp, 0), 
                                                MAKEARRAY( 1, Periods, 
                                                    LAMBDA(R, C, 
                                                            LET(Idx, (C - 1) * MpP + 1,
                                                                SUM( CHOOSECOLS(MoInt, SEQUENCE(1, MpP, Idx)))
                                                            )
                                                    )
                                                )
                                            )
                                        ),    
                    PaymentArray,       LET(MoPmt,  EXPAND( SEQUENCE(1, Term, MonthlyPayment, 0), 1, ROUNDUP( Term/ Mpp, 0) * Mpp, 0), 
                                            MAKEARRAY( 1, Periods, 
                                                LAMBDA(R, C, 
                                                        LET(Idx, (C - 1) * MpP + 1,
                                                            SUM( CHOOSECOLS(MoPmt, SEQUENCE(1, MpP, Idx)))
                                                        )
                                                )
                                            )

                                        ), 
                    ClosingArray,       PrincipalArray + InterestArray + PaymentArray,
                    PortionArray,       -PaymentArray - InterestArray,    
                    AmortisationSched,  VSTACK(
                                            DebtIssueArray, 
                                            PrincipalArray, 
                                            InterestArray, 
                                            PaymentArray, 
                                            ClosingArray, 
                                            PortionArray),
                //  Position in timeline
                    Offset,             TimelineOffsetλ(StartDate, Timeline),
                //  A sub-monthly period is a fixed number of days, so each month lands in
                //  the one period that contains its start and the periods between are nil.
                    MonthStarts,        EDATE( StartDate, SEQUENCE( , Periods, 0)),
                //  The period each month opens in is the last period start on or before it. A
                //  period ends where the next one opens; only the last has no successor to ask,
                //  so it runs on as far as the period before it did. A month before the first
                //  period or after that end belongs to no period and lands nowhere.
                    Buckets,            IFERROR( MATCH( MonthStarts, Timeline, 1), 0)
                                        * ( MonthStarts < MAX( Timeline) + LastGap),
                    InPeriod,           ( TOCOL( Buckets) = SEQUENCE( 1, COUNTA( Timeline))) * 1,
                //  Rows 2 and 5 are balances rather than flows. A period can hold several
                //  months, as the last one does when FinalPeriodEnd extends it, or a longer
                //  period after shorter ones: the flows of its months add up, but it opens on
                //  its first month's balance and closes on its last month's, which is how
                //  Depreciateλ reads its own opening balance row. Summed with the flows, a loan
                //  repaid inside such a period closed it on the total of its monthly balances.
                    Earlier,            IF( Periods = 1, 0, HSTACK( 0, DROP( Buckets, , -1))),
                    Later,              IF( Periods = 1, 0, HSTACK( DROP( Buckets, , 1), 0)),
                    FirstInPeriod,      ( Buckets <> Earlier) * ( Buckets > 0),
                    LastInPeriod,       ( Buckets <> Later) * ( Buckets > 0),
                    Flows,              MMULT( AmortisationSched, InPeriod),
                    ByDate,             VSTACK( TAKE( Flows, 1),
                                                MMULT( CHOOSEROWS( AmortisationSched, 2) * FirstInPeriod, InPeriod),
                                                CHOOSEROWS( Flows, 3, 4),
                                                MMULT( CHOOSEROWS( AmortisationSched, 5) * LastInPeriod, InPeriod),
                                                TAKE( Flows, -1)),
                    IF( SubMonthly?,
                        ByDate,
                        TimelinePositionλ(AmortisationSched, Timeline, Offset))
                )   
            ),    
    //  Every loan in turn, a blank row between one block and the next
        Result,     REDUCE( 0, SEQUENCE( ROWS( Principals)),
                        LAMBDA( Acc, Asset,
                            LET(NewBlock,   Schedule( Asset),
                                IF( Asset = 1, 
                                    NewBlock,

//  Depreciation Suite

/*  FUNCTION NAME:  Depreciateλ
    DESCRIPTION:*//**Create a block of CAPEX, Opening Balance, Depreciation Values, and Book Value for each asset*/
Depreciateλ = LAMBDA(
//  Parameter Declarations
    [InitialValues],
    [InServiceDates], 
    [LifeInYears], 
    [Timeline],
    [SalvageValues],
    [DisposalDates],
    [DisposalCosts],
    [Methods],
    [Factors],
    LET(
//  Help
        Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →Depreciateλ(InitialValues, InServiceDates, LifeInYears, Timeline, ¶" &
                            "               →[SalvageValues], [DisposalDates], [DisposalCosts], [Methods], [Factors])¶" &
                            "DESCRIPTION:   →Creates an array of CAPEX, depreciation, and book values for a model's timeline¶" & 
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit ¶" & 
                            "VERSION:       →20 Aug 2026¶" &
                            "NOTES:         →The methods used here differ from Excel's functions in several ways. Please see webpage for details.¶" &
                            "PARAMETERS:    →¶" & 
                            "InitialValues  →(Required) The purchase values of each asset to depreciate¶" & 
                            "InServiceDates →(Required) The dates from which to start depreciating each asset¶" & 
                            "LifeInYears    →(Required) The whole number of years over which to depreciate each asset.¶" & 
                            "               →More than 0 and no more than 100. A date here is not a life:¶" & 
                            "               →the arguments read InitialValues, InServiceDates, LifeInYears, Timeline.¶" & 
                            "Timeline       →(Required) The model's timeline or a row of period start dates. ¶" & 
                            "               →Timeline can be any interval: days, weeks, months, quarters or years¶" & 
                            "SalvageValues  →(Optional: Default = 0) The book values of each asset after fully depreciated.¶" & 
                            "DisposalDates  →(Optional: Default = End of life) The dates on which the assets are removed from the books.¶" & 
                                           "→A disposal before the end of life writes the remaining book value off in that month.¶" & 
                            "DisposalCosts  →(Optional: Default = 0) The costs of removing each asset from the books, charged in the disposal period.¶" & 
                            "Methods        →(Optional: Default = SLN) The desired depreciation method. Must be one of these method codes: ¶" & 
                                           "→SLN=Straight Line Depreciation¶" & 
                                           "→SYD=Sum of Year's Digits¶" & 
                                           "→DB =Declining Balance¶" & 
                                           "→DDB=Double Declining Balance¶" & 
                                           "→VDB=Variable Declining Balance¶" & 
                                           "→DV =Diminishing balance at 200% of straight line. Salvage value ignored¶" & 
                                           "→PC =Straight line, whole years. Salvage value ignored¶" & 
                            "Factors        →(Optional: Default = 2) Only applicable to DDB and VDB methods.", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( InitialValues),
                            ISOMITTED( LifeInYears),
                            ISOMITTED( InServiceDates),
                            ISOMITTED( Timeline)
                        ),
    //  Set Constants
        MethodArray,    {"SLN","SYD","DB","DDB","VDB","DV","PC"},
        Mpy,            12, //Months Per Year
    //  Check inputs - a life in years must be a life, not a date
        LifeGiven,      IF( ISOMITTED( LifeInYears), 1, LifeInYears),
        LifeNums,       IFERROR( VALUE( LifeGiven), -1),
        BadLife?,       OR( MIN( LifeNums) <= 0, MAX( LifeNums) > 100,
                            SUMPRODUCT( --( MOD( LifeNums, 1) <> 0)) > 0),
        LifeMessage,    "LifeInYears must be a whole number of years greater than 0 and no more than 100. "
                        & "Check the argument order: InitialValues, InServiceDates, LifeInYears, Timeline.",
    //  Set defaults and apply convenience conversions
        SalvageValues,  IF( ISOMITTED( SalvageValues), 
                            MAP( InitialValues, LAMBDA( Ele, 0)), 
                            MAP( SalvageValues, LAMBDA(v, IFERROR( VALUE( v), 0)))),
        CvtSrvDates,    IF( NOT( ISOMITTED( InServiceDates)), 
                            IF( ISNUMBER( InServiceDates), 
                                InServiceDates, 
                                DATEVALUE( InServiceDates))),
        DisposalDates,  IF( ISOMITTED( DisposalDates), 
                            EDATE( +CvtSrvDates, LifeinYears * 12), 
                            MAKEARRAY( ROWS( DisposalDates), 1, 
                                LAMBDA(R, C, 
                                    IF( INDEX( DisposalDates, R)  = "", 
                                        EDATE( INDEX( CvtSrvDates, R), INDEX(LifeinYears, R) * 12), 
                                        INDEX( DisposalDates, R)
                                    )
                                )
                            )
                        ),
        DisposalCosts,  IF( ISOMITTED( DisposalCosts), 
                            MAP( InitialValues, LAMBDA( Ele, 0)), 
                            DisposalCosts),
        Methods,        IF( ISOMITTED( Methods), 
                            MAP( InitialValues, LAMBDA( Ele, "SLN")), 
                            MAP( Methods, LAMBDA(v, IF( TRIM( v) = "", "SLN", v)))),
        Factors,        IF( ISOMITTED( Factors), 
                            MAP( InitialValues, LAMBDA( Ele, 2)), 
                            MAP( Factors, LAMBDA(v, IF( TRIM( v) = "", 2, v)))),
    //  Create individual Values from timeline array
        TimelineCols,   COLUMNS( TImeline),
        FirstPeriod,    INDEX( Timeline, 1),
        SecondPeriod,   INDEX( TImeline, 2),
        DpP,            SecondPeriod - FirstPeriod,                      //Days Per Period
        SubMonthly?,    DpP < 28,                                        //No month is shorter
        MpP,            MAX( @ROUND( DpP / 30.5, 0), 1),                 //Months Per Period, never nought
    //  The last period has no successor to take its end date from, so it ends one period
    //  on from its own start: a whole number of months where the timeline is monthly or
    //  longer, the same number of days as every other period where it is shorter.
    //  A month-end start steps forward through EOMONTH, because EDATE(28 Feb, 1) is
    //  28 Mar, not 31 Mar, and the three days between fall out of the schedule. This is
    //  the step DayCountRateλ() already takes for its own last period.
        LastGap,        MAX( Timeline) - INDEX( Timeline, TimelineCols - 1),
        LastDate,       MAX( Timeline),
        LastEnd,        IF( SubMonthly?,
                            LastDate + LastGap,
                            IF( DAY( LastDate + 1) = 1,
                                EOMONTH( LastDate, MpP),
                                EDATE( LastDate, MpP))) - 1,
        EndDates,       HSTACK( MAP(  SEQUENCE(, TimelineCols - 1), LAMBDA( n, INDEX( Timeline, n + 1) - 1)), LastEnd),
    //  Process each asset
        Result,         REDUCE( 0, SEQUENCE( ROWS( InitialValues)), 
                            LAMBDA(Block, Asset, 
                                LET( 
                                //  Create individual values from arguments
                                    Method,         @INDEX( Methods, Asset),
                                    Years,          IF( BadLife?, 1, @INDEX( LifeInYears, Asset)),
                                    InitialValue,   @INDEX( InitialValues, Asset), 
                                    SalvageValue,   IF( OR( Method = "DV", Method = "PC"), 0, @INDEX( SalvageValues, Asset)), 
                                    InServiceDate,  @INDEX( CvtSrvDates, Asset), 
                                    DisposalDate,   @INDEX( DisposalDates, Asset), 
                                    DisposalCost,   @INDEX( DisposalCosts, Asset), 
                                    Factor,         @INDEX( Factors, Asset),
                                    LifeInMonths,   PeriodDiffλ( InserviceDate, DisposalDate, "M"),
                                //  Procedure
                                //  Depreciation is calculated annually then allocated equally over the next 12 months
                                    Depreciate,     CHOOSE( MATCH( @Method, MethodArray, 0), //CHOOSE is more efficient than SWITCH
                                                    //  1. Straight line 
                                                        SLNλ( InitialValue, SalvageValue, Years), 
                                                    //  2. Sum-of-years' digits
                                                        SYDλ( InitialValue, SalvageValue, Years),
                                                    //  3. Declining balance
                                                        DBλ( InitialValue, SalvageValue, Years),
                                                    //  4. Double declining balance 
                                                        DDBλ( InitialValue, SalvageValue, Years, Factor),
                                                    //  5. Variable declining balance 
                                                        VDBλ( InitialValue, SalvageValue, Years, Factor), 
                                                    //  6. Diminishing balance at 200% of straight line
                                                        DiminishingValueλ( InitialValue, Years),
                                                    //  7. Straight line, whole years
                                                        PrimeCostλ( InitialValue, Years),
                                                    ),
                                //  Allocating annual values to months. The asset's months run to its disposal:
                                //  a disposal after the end of life pads the tail with nought and writes the
                                //  salvage value off in the last month; a disposal before the end of life stops
                                //  the schedule there and writes the remaining book value off instead.
                                    Monthly,        Allocateλ( Depreciate, "Y", "M"),
                                    Months,         LifeInMonths - 1,
                                    Allocate,       IF( Months < 1,
                                                        InitialValue,
                                                        IF( Months >= COLUMNS( Monthly),
                                                            HSTACK( EXPAND( Monthly, 1, Months, 0), SalvageValue),
                                                            HSTACK( TAKE( Monthly, , Months), InitialValue - SUM( TAKE( Monthly, , Months))))),
                                    Opening,        DROP( HSTACK( InitialValue, SCAN( InitialValue, Allocate, LAMBDA( Acc, Val, Acc - Val))), 0, -1),
                                //  Create a monthly timeline for the asset
                                    AssetTimeline,  EDATE( InServiceDate, SEQUENCE( , LifeInMonths, 0)), 
                                //  The period each asset month falls in is the last period start on or before it;
                                //  a month after the last period ends belongs to no period. One matrix product
                                //  then aggregates the months into the model's periods.
                                    Buckets,        IFERROR( MATCH( AssetTimeline, Timeline, 1), 0) * ( AssetTimeline <= LastEnd),
                                    InPeriod,       ( TOCOL( Buckets) = SEQUENCE( 1, TimelineCols)) * 1,
                                    Depreciation,   MMULT( Allocate, InPeriod),
                                    CAPEX,          InitialValue * (Timeline <= InServiceDate) * (EndDates >= InServiceDate), 
                                //  The opening balance a period shows is the one its first asset month carries
                                    Earlier,        IF( LifeInMonths = 1, 0, HSTACK( 0, DROP( Buckets, , -1))),
                                    FirstInPeriod,  ( Buckets <> Earlier) * ( Buckets > 0),
                                    OpeningAmount,  MMULT( Opening * FirstInPeriod, InPeriod),
                                    BookValue,      OpeningAmount - Depreciation,
                                    SalesRevenue,   SalvageValue * (Timeline <= DisposalDate) * (EndDates >= DisposalDate), 
                                    DisposalExp,    DisposalCost * (Timeline <= DisposalDate) * (EndDates >= DisposalDate), 
                                //  Assemble this assets rows into a block
                                    NewBlock,       VSTACK( CAPEX, OpeningAmount, Depreciation, BookValue, SalesRevenue, DisposalExp),
                                //  Add this block to prior blocks (if any)
                                    Result,         IF( Asset = 1, 
                                                        NewBlock, 
                                                        VSTACK(
                                                            Block,
                                                            EXPAND("", 1, TimelineCols, ""),
                                                            NewBlock
                                                            )
                                                    ),
                                    Result
                                )
                            )
                        ),
    //  Return Result, Help, or the message
        Return,         IF( Help?, 2, IF( BadLife?, 3, 1)),
        CHOOSE( Return, Result, Help, LifeMessage)
    ) 
);

/*  FUNCTION NAME:  Allocateλ
    DESCRIPTION:*//**→Divide one or more amounts equally across smaller periods*/
Allocateλ = LAMBDA(
//  Parameter Declaration
    [Amounts],
    [FromInterval], 
    [ToInterval], 
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →Allocateλ(Amounts, [FromInterval], [ToInterval])¶" &
                            "DESCRIPTION:   →Divide one or more amounts equally across smaller periods¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                            "VERSION:       →20 Aug 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Amounts        →(Required) One amount or a row of amounts to allocate.¶" &                   
                            "FromInterval   →(Optional Default=Y) The larger interval in Q=Quarters or Y=Years¶" & 
                            "ToInterval     →(Optional Default=M) The smaller interval in W=Weeks, M=Months, or Q=Quarters¶" & 
                            "NOTE!          →¶" & 
                                           "→When a From Period's amount does not divide evenly to the 1/100 for all To Periods',¶" & 
                                           "→the last To Period contains the necessary adjustment.¶" & 
                            "EXAMPLES:      →¶" &
                                           "→Formula (oz is assumed to be the module's name)¶" &
                                           "→=oz.Allocateλ({999.99, 2000}, ""Y"", ""Q"")¶" &
                                           "→Result¶" &
                                           "→250,250,250,249.99,500,500,500,500", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          ISOMITTED( Amounts),
    //  Set defaults for optional values
        FromInterval,   IF( ISOMITTED( FromInterval), "Y", FromInterval),
        ToInterval,     IF( ISOMITTED( ToInterval), "M", ToInterval),
    //  Procedure
        From,           SWITCH(FromInterval, "Y", 1, "Q", 4),
        To,             SWITCH(ToInterval, "Q", 4, "M", 12, "W", 52),
        FromCount,      COUNT( Amounts),
        ToCount,        To / From,
        Result,         MAKEARRAY( 1, FromCount * ToCount,
                            LAMBDA( R, C,
                                LET(
                                //  Column C holds part C of amount QUOTIENT( C - 1, ToCount) + 1
                                    FromAmount,     INDEX( Amounts, QUOTIENT( C - 1, ToCount) + 1),
                                    ToAmount,       ROUND( FromAmount * From / To, 2),
                                //  The last part of each amount carries the rounding difference
                                    Last?,          MOD( C - 1, ToCount) + 1 = ToCount,
                                    IF( AND( Last?, ToCount > 1),
                                        FromAmount - SUM( EXPAND( ToAmount, 1, MAX( ToCount - 1, 1), ToAmount)),
                                        ToAmount)
                                )
                            )
                        ),
        
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);



/*  FUNCTION NAME:  PeriodDiffλ
    DESCRIPTION:*//**Determine the number of periods between two dates.*/
PeriodDiffλ = LAMBDA(
//  Parameter Declaration
    [StartDate],
    [EndDate], 
    [Interval], 
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →PeriodDiffλ(StartDate, EndDate, Interval)¶" &
                            "DESCRIPTION:   →Determine the number of periods between two dates.¶" &
                            "NOTES!         →This improves on, and corrects problems w/Excel's deprecated DATEDIF() function.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                            "VERSION:       →20 Aug 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "StartDate      →(Required) Start Date.¶" &                   
                            "EndDate        →(Required) End Date¶" & 
                            "Interval       →(Required) W=Week, M=Month, Q=Quarter, Y=Year.¶" & 
                            "EXAMPLES:      →¶" &
                            "Result         →Formula (oz is assumed to be the module's name)¶" &
                            "2              →=oz.PeriodDiffλ(""2026-01-01"", ""2026-04-15"", ""Q"")", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( StartDate),
                            ISOMITTED( EndDate),
                            ISOMITTED( Interval)
                        ),
    //  Constants
        MpP,                SWITCH( Interval, "M", 1, "Q", 3, "Y", 12, 0),
    //  Procedure
        Months,             DATEDIF( StartDate, EndDate, "M") + 1,
    //  MpP is 0 for "W" and for any interval SWITCH does not recognise, and both
    //  INT( Months / 0) and MOD( Months, 0) are #DIV/0!. The weekly result never
    //  needs the month arithmetic, and an unrecognised interval is reported below.
        Divisor,            IF( MpP = 0, 1, MpP),
        Periods,            INT( Months / Divisor),
        PartialPeriod,      N( MOD( Months, Divisor) > 0),
        EOMCorrection,      N( AND( EndDate = EOMONTH( EndDate, 0), DAY( StartDate) > Day( EndDate))),
        Result,             IFS( Interval = "W", INT((EndDate - StartDate + 1) / 7),
                                 MpP = 0,        NA(),
                                 TRUE,           Periods + OR(PartialPeriod, EOMCorrection)),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);



/*  FUNCTION NAME:  SLNλ
    DESCRIPTION:*//**Straight Line depreciation method for one asset or asset class.*/
SLNλ = LAMBDA(
//  Parameter Declaration
    [Cost],
    [Salvage], 
    [Life], 
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →SLNλ(Cost, Salvage, Life)¶" &
                            "DESCRIPTION:   →Straight Line Depreciation Method for one asset or asset class.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                            "VERSION:       →20 Aug 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Cost           →(Required) Asset's initial cost.¶" &                   
                            "Salvage        →(Required) Asset's value and end of life¶" & 
                            "Life           →(Required) Asset's useful life in years.¶" & 
                            "EXAMPLES:      →¶" &
                                           "→Formula (oz is assumed to be the module's name)¶" &
                                           "→=oz.SLNλ(1000, 100, 5)¶" &
                                           "→Result¶" &
                                           "→180,180,180,180,180", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Cost),
                            ISOMITTED( Salvage),
                            ISOMITTED( Life)
                        ),
    //  Procedure. A fractional life keeps its whole years at the annual charge and
    //  puts the remainder in one final part-year column, so nothing is left undepreciated.
        Depn,           SLN( Cost, Salvage, Life),
        Whole,          INT( Life),
        Result,         IF( Whole = Life, EXPAND( Depn, 1, Life, Depn),
                            IF( Whole = 0, Cost - Salvage,
                                HSTACK( EXPAND( Depn, 1, Whole, Depn), Cost - Salvage - Depn * Whole))),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  AmortiseλDV
    DESCRIPTION:*//**Provides extended data validation of arguments for Amortiseλ()*/
AmortiseλDV = LAMBDA(
//  Parameter Declarations
    [Principal],
    [APR],
    [Term], 
    [StartDate],
    [Timeline], 
    [FinalPeriodEnd],

    LET(
    //  Check inputs - Errors in argument values
        ErrorsInArgs,   VSTACK( 
                            OR( ISERROR( Principal)),
                            OR( ISERROR( APR)),
                            OR( ISERROR( Term)),
                            OR( ISERROR( StartDate)),
                            OR( ISERROR( Timeline)),
                        //  Amortiseλ reads this one through MAX, so an error in it stops a
                        //  schedule and has to stop the validator too. An omitted argument
                        //  is not an error, and ISERROR cannot be asked about one.
                            IF( ISOMITTED( FinalPeriodEnd), FALSE, OR( ISERROR( FinalPeriodEnd)))
                        ),

    //  Check inputs - Data validation errors
        DVArg1,         IF( OR( ISOMITTED( Principal), ISERROR( Principal)), 
                            FALSE, 
                            OR( IFERROR( VALUE(Principal) < 1, TRUE))),
        DVArg2,         IF( OR( ISOMITTED( APR), ISERROR( APR)), 
                            FALSE, 
                            OR( IFERROR( VALUE(APR) < 0, TRUE))),
        DVArg3,         IF( OR( ISOMITTED( Term), ISERROR( Term)), 
                            FALSE, 
                            OR( IFERROR( VALUE(Term) < 1, TRUE))),
        DVArg4,         IF( OR( ISOMITTED( StartDate), ISERROR( StartDate)), 
                            FALSE, 
                            OR( IFERROR( VALUE(StartDate) < MIN( Timeline), TRUE))),
        DVArg5,         IF( OR( ISOMITTED( Timeline), ISERROR( Timeline)), 
                            FALSE, 
                            OR( IFERROR( VALUE(Timeline) < 1, TRUE))),
        DVArg6,         IF( OR( ISOMITTED( Timeline), ISERROR( Timeline), DVArg5), 
                            FALSE, 
                            COUNTA(Timeline) < 2),
    //  Amortiseλ reads a gap under 28 days as a sub-monthly timeline and lays the
    //  schedule out in days, so only a timeline that does not advance is wrong here.
    //  Every consecutive pair is measured. The first gap alone let a repeated date
    //  later in the row through, and the parent's approximate MATCH then buckets both
    //  copies into one period and drops the duplicate's value.
        DVArg7,         IF( OR( ISOMITTED( Timeline), ISERROR( Timeline), DVArg5, DVArg6), 
                            FALSE, 
                            LET( Dates, TOROW( Timeline),
                                 MIN( DROP( Dates, , 1) - DROP( Dates, , -1)) < 1)),
    //  FinalPeriodEnd is the day after the last period ends, so it falls after the last
    //  date in the timeline. COUNT is 1 for one date, 0 for text and more than 1 for a

    //  without a timeline rather than measure it against the default it chose for itself,
    //  so this refuses the same call rather than passing one its parent will not answer.
        DVArg8,         IF( OR( ISOMITTED( FinalPeriodEnd), ISERROR( FinalPeriodEnd)),
                            FALSE,
                            OR( ISOMITTED( Timeline),
                                COUNT( FinalPeriodEnd) <> 1,
                                IF( OR( ISERROR( Timeline), DVArg5),
                                    FALSE,
                                    MAX( FinalPeriodEnd) <= MAX( Timeline)))),
        DVErrors,       VSTACK( DVArg1, DVArg2, DVArg3, DVArg4, DVArg5, DVArg6, DVArg7, DVArg8),
        
    //  Any input errors detected?
        AnyErrors?,     OR(ErrorsInArgs, DVErrors),

    //  Assemble Error Messages Array (if errors found)
        Errors2Show,    VSTACK( ErrorsInArgs, DVErrors), 
        ErrMsgs,        {
                    //  Errors in arguments
                        "Amount contains errors in cells. Resolve errors first.";
                        "APR contains errors in cells. Resolve errors first.";
                        "Term contains errors in cells. Resolve errors first.";
                        "StartDate contains errors in cells. Resolve errors first.";
                        "Timeline contains errors in cells. Resolve errors first.";
                        "FinalPeriodEnd contains errors in cells. Resolve errors first.";

                    //  Data validation errors
                        "Amount must be greater than 0";
                        "APR must be a percentage of 0 or more";
                        "Term must be the number of timeline periods until the debt is satisfied.";
                        "StartDate must be a valid date.";
                        "Timeline must a row of period start dates.";
                        "Timeline must a row of period start dates with more than 1 entry.";
                        "Timeline must a row of period start dates that advance by at least a day.";
                        "FinalPeriodEnd needs a Timeline, and must be one date after the last date in it."},
        Messages,       FILTER(ErrMsgs, Errors2Show, ""),        

    //  Handle Error
        Return,         IF( OR( Errors2Show), 2, IF( AnyErrors?, 3, 1)),      
    //  Return Result
        CHOOSE(Return, TRUE, Messages, #VALUE!)
    )    
);

/*  FUNCTION NAME:  TimelineOffsetλ
    DESCRIPTION:*//**Determines how many periods one array is offset from the timeline*/
/*  DISCUSSION:     The possibility exists requiring calculations before a model's timeline. An example is
                    depreciation of assets in an existing company wanting to forecast some activity. 
                    In such cases, it can be advantageous to perform the calculations within their own timeline
                    and then place the results properly within the model's timeline. 
                    This routine determine where such calculation results fit within the model's timeline and 
                    crops those results that do not belong. */
TimelineOffsetλ = LAMBDA(
//  Parameter Declarations
    [Date],    
    [Timeline],      
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →TimelineOffsetλ(Date, Timeline)¶" &
                            "DESCRIPTION:   →Determines in which period, relative to the timeline's start and interval,¶" &
                                           "→a date falls within.¶" & 
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                            "VERSION:       →20 Aug 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Date           →(Required) A starting date¶" & 
                            "Timeline       →(Required) A model's timeline (Row of period start dates)¶" &                   
                            "EXAMPLES:      →¶" &
                            "Result         →Formula (oz is assumed to be the module's name)¶" &
                            "1              →=oz.TimelineOffsetλ(""15/8/2026"", EDATE(""1/7/2026"", SEQUENCE( , 12, 0)))¶" &
                            "-11            →=oz.TimelineOffsetλ(""15/8/2025"", EDATE(""1/7/2026"", SEQUENCE( , 12, 0)))", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Date),
                            ISOMITTED( Timeline)
                        ),
    //  Check inputs - If date is in text, convert to value
        Date,           @IF( ISTEXT( Date), DATEVALUE( Date), Date),
    /*  Premise:        Create a short timeline that is in sync with the model's timeline that spans 
                        from the timeline start to the date in question plus one period on either side.
                        No we can use MATCH() to find which timeline period    */
    //  Determine timeline's interval (length of periods). A one-date timeline has no
    //  second date to measure, so it is read as monthly.
        Period1,        INDEX( Timeline, 1),
        Period2,        IF( COUNTA( Timeline) < 2, EDATE( Period1, 1), INDEX( Timeline, 2)),
        DpP,            Period2 - Period1,                    //Days Per Period
        MpP,            MAX( ROUND( DpP/30.5, 0), 1),         //Months Per Period, never zero
        SubMonthly?,    ROUND( DpP/30.5, 0) = 0,
    //  Determine if Prior to timeline or after
        Direction,      IF(Date < Period1, -1, 1),
    //  Determine the number of months between timeline's start and the date's period start
        MonthDiff,      @DATEDIF( MIN( Period1, Date), MAX( Period1, Date), "M"),
    //  Convert to number of periods
        PeriodDiff,     QUOTIENT( MonthDiff, MpP),
        PeriodSeq,      SEQUENCE( 1, PeriodDiff + 2, IF( Direction = 1, 0, -( PeriodDiff + 1))),
        SearchTimeline, EDATE( Period1, PeriodSeq * MpP),
        ByMonth,        MATCH( Date, SearchTimeline, 1) - IF( Direction = 1, 1, PeriodDiff + 2),
    //  A sub-monthly timeline is a fixed number of days per period, so the offset is the
    //  whole number of periods between the timeline start and the date, floored, which
    //  puts a date before the timeline in the period that ends where the timeline begins.
        Result,         IF( SubMonthly?, INT( (Date - Period1) / DpP), ByMonth),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  PeriodStartλ
    DESCRIPTION:*//**Determine when the period containing a date of interest starts*/
PeriodStartλ = LAMBDA(
//  Parameter Declaration
    [AnyPeriodStart],
    [MonthsPerPeriod], 
    [DateOfInterest], 
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →PeriodStartλ(AnyPeriodStart, MonthsPerPeriod, DateOfInterest)¶" &
                            "DESCRIPTION:   →Determine when the period containing a date of interest starts.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" & 
                            "VERSION:       →20 Aug 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "AnyPeriodStart →(Required) Any start date within a timeline.¶" &                   
                            "MonthsPerPeriod→(Required) Number of months in each period¶" & 
                            "DateOfInterest →(Required) A date for which we want to know the start of the period containing it.¶" & 
                            "EXAMPLES:      →¶" &
                            "Result         →Formula (oz is assumed to be the module's name)¶" &
                            "2026-10-01     →=oz.PeriodStartλ(""2026-07-01"", 3, ""2026-10-15"")", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( AnyPeriodStart),
                            ISOMITTED( MonthsPerPeriod),
                            ISOMITTED( DateOfInterest)
                        ),
    //  Constants
        MpY,                12, //Months Per Year
        MpP,                MonthsPerPeriod,
    //  Check inputs - If a date is in text, convert to value
        CvtStart,           IF( ISTEXT( AnyPeriodStart), DATEVALUE( AnyPeriodStart), AnyPeriodStart),
        CvtDate,            IF( ISTEXT( DateOfInterest), DATEVALUE( DateOfInterest), DateOfInterest),
    //  Procedure
    //  Count the whole periods from the anchor to the date of interest and step the
    //  anchor on by that many.
        MonthDiff,          (YEAR( CvtDate) - YEAR( CvtStart)) * MpY + MONTH( CvtDate) - MONTH( CvtStart),
        WholePeriods,       QUOTIENT( MonthDiff, MpP),
        Candidate,          EDATE( CvtStart, WholePeriods * MpP),
    //  QUOTIENT truncates towards zero, and the day of the month can leave the candidate
    //  past the date of interest, so step back a period where it does. One step is always
    //  enough: a truncated quotient is never more than one period out.
        Result,             IF( Candidate > CvtDate, EDATE( CvtStart, (WholePeriods - 1) * MpP), Candidate),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                            "VERSION:       →20 Aug 2026¶" &

/*  FUNCTION NAME:  LeaseScheduleλ
    DESCRIPTION:*//**Unwinds an AASB 16 lease liability: opening, payment, interest and closing for each period*/

LeaseScheduleλ = LAMBDA(
//  Parameter Declaration
    [Payments],
    [Rate],
    [InAdvance],
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:      →LeaseScheduleλ(Payments, Rate, [InAdvance])¶" &
                            "DESCRIPTION:   →Unwinds an AASB 16 lease liability over its term.¶NOTES!         →Returns four rows: opening liability, payment, interest, closing¶               →liability. When InAdvance is TRUE, the first supplied payment is paid¶               →at the measurement date and is excluded from both the opening liability¶               →and the payment row. The¶               →opening balance is LeaseLiabilityλ() on the same arguments.                             "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                            "VERSION:       →24 Aug 2026¶" &
                            "PARAMETERS:    →¶" &
                            "Payments       →(Required) One row or one column of lease payments, one per period, entered positive. Include the measurement-date payment first when InAdvance is TRUE.¶" &
                            "Rate           →(Required) Discount rate per period, not per year.¶" &
                            "InAdvance      →(Optional: Default = FALSE) TRUE when the first supplied payment is paid at the measurement date.¶" &
                            "EXAMPLES:      →¶" &
                            "→Formula (oz is assumed to be the module's name)¶" &
                            "→=oz.LeaseScheduleλ({100,100,100}, 0.05)¶" &
                            "→Result¶" &
                            "→272.32, 185.94,  95.24¶" &
                            "→100.00, 100.00, 100.00¶" &
                            "→ 13.62,   9.30,   4.76¶" &
                            "→185.94,  95.24,   0.00",
                            "→", "¶"
                        )),
    //  Check inputs - Omitted required arguments
        Help?,          OR(ISOMITTED(Payments), ISOMITTED(Rate)),
    //  A blank InAdvance cell is not an omitted argument, so test for both
        Advance?,       IF(OR(ISOMITTED(InAdvance), TRIM(InAdvance & "")=""), FALSE, InAdvance),
    //  One lease at a time, laid along a row the way the timeline helpers expect
        Flat,           IF(ROWS(Payments) = 1, Payments, TRANSPOSE(Payments)),
        InputCount,     COLUMNS(Flat),
        LiabilityPayments, IF(AND(Advance?, InputCount > 1),
                                DROP(Flat, , 1),
                                IF(Advance?, 0, Flat)),
        Count,          COLUMNS(LiabilityPayments),
        Opening1,       LeaseLiabilityλ(Flat, Rate, Advance?),
    //  Every payment left in the liability vector follows one period of interest
        Closings,       SCAN(Opening1, LiabilityPayments, LAMBDA(Balance, Payment,

    //  Each opening is the closing before it. A one-period lease has no closing to shift.
        Openings,       IF(Count = 1, Opening1, HSTACK(Opening1, DROP(Closings, , -1))),

        Result,         VSTACK(Openings, LiabilityPayments, Interest, Closings),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

                            ISOMITTED(CarryingLiability), ISOMITTED(CarryingROU)),
    //  A blank InAdvance cell is not an omitted argument, so test for both

        Adjustment,     Revised - CarryingLiability,
    //  The asset absorbs the adjustment until it is exhausted, and no further
        RawAsset,       CarryingROU + Adjustment,
        Result,         VSTACK(Revised, Adjustment, MAX(RawAsset, 0), MIN(RawAsset, 0)),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )

/*  FUNCTION NAME:  PeriodRateλ
    DESCRIPTION:*//**Converts an effective annual rate to the equivalent rate per period, the rate the lease functions take*/

PeriodRateλ = LAMBDA(
//  Parameter Declaration
    [AnnualRate],
    [PeriodsPerYear],
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:      →PeriodRateλ(AnnualRate, [PeriodsPerYear])¶" &
                            "DESCRIPTION:   →Converts an effective annual rate to the equivalent rate per period.¶NOTES!         →(1 + AnnualRate) ^ (1 / PeriodsPerYear) - 1, so compounding the result¶               →PeriodsPerYear times gives the annual rate back. This is the rate per¶               →period that LeaseLiabilityλ(), LeaseScheduleλ() and LeaseRemeasureλ()¶               →take. It is for an effective annual rate. A nominal annual percentage¶               →rate that compounds monthly is already APR / 12 per month, so do not¶               →pass it here; EFFECT() turns one into the effective rate this takes.¶               →AnnualRateλ() is the inverse.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                            "VERSION:       →6 Sep 2026¶" &
                            "PARAMETERS:    →¶" &
                            "AnnualRate     →(Required) Effective annual rate, or a row or column of them.¶" &
                            "PeriodsPerYear →(Optional: Default = 12) Periods in a year: 12 for months, 4 for quarters, 1 for years.¶" &
                            "EXAMPLES:      →¶" &
                            "→Formula (oz is assumed to be the module's name)¶" &
                            "→=oz.PeriodRateλ(0.05)¶" &
                            "→Result¶" &
                            "→0.004074",
                            "→", "¶"
                        )),
    //  Check inputs - Omitted required arguments
        Help?,          ISOMITTED(AnnualRate),
    //  A blank PeriodsPerYear cell is not an omitted argument, so test for both. The
    //  blank test stays element-wise: OR() reduced a column of frequencies to one TRUE,
    //  so a single blank gave every row 12.
        Periods,        IF(ISOMITTED(PeriodsPerYear), 12,
                            IF(ISNUMBER(PeriodsPerYear), PeriodsPerYear, 12)),
        Result,         (1 + AnnualRate) ^ (1 / Periods) - 1,
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  DayCountRateλ
    DESCRIPTION:*//**Interest rate for each period of a timeline under a day count convention: 30/360, Actual/360, Actual/365 or Actual/Actual*/

DayCountRateλ = LAMBDA(
//  Parameter Declaration
    [Timeline],
    [APR],
    [Convention],
    [EndDates],
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:      →DayCountRateλ(Timeline, APR, [Convention], [EndDates])¶" &
                            "DESCRIPTION:   →Interest rate for each period of a timeline under a day count convention.¶NOTES!         →Returns a row with one rate per timeline period, whichever way the¶               →timeline runs: APR multiplied by the days in the period and divided by¶               →the days in the year, as the convention counts them. Pass the result to¶               →DebtSculptVariableλ() or DebtSculptVariableLRVλ() as PeriodRates in place¶               →of a flat twelfth of the APR. With start dates each period runs from its¶               →date to the day before the next, and the last period takes its length¶               →from the one before it; with end dates each runs from the day after the¶               →previous date to its own, and the first period is the one inferred. A¶               →gap of 28 days or more is read as whole calendar months, a shorter one¶               →as days. 30/360 counts every month as 30 days on the European rule.¶               →Under Actual/Actual each calendar year uses its own length as the¶               →denominator (the ISDA rule), including periods spanning several years.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                            "VERSION:       →10 Sep 2026¶" &
                            "PARAMETERS:    →¶" &
                            "Timeline       →(Required) A row or column of period dates, at least two, evenly spaced.¶" &
                            "APR            →(Required) Annual percentage rate, one value or one per period.¶" &
                            "Convention     →(Optional: Default = 3) 1 = 30/360, 2 = Actual/360, 3 = Actual/365, 4 = Actual/Actual. Any other number is #VALUE!.¶" &
                            "EndDates       →(Optional: Default = FALSE) TRUE, or a number other than 0, when the timeline shows period end dates rather than start dates.¶" &
                            "EXAMPLES:      →¶" &
                            "→Formula (oz is assumed to be the module's name)¶" &
                            "→=oz.DayCountRateλ(EDATE(DATE(2026,7,1), {0,1,2}), 0.073)¶" &
                            "→Result¶" &
                            "→0.0062, 0.0062, 0.0060",
                            "→", "¶"
                        )),
    //  Check inputs - Omitted required arguments
        Help?,          OR(ISOMITTED(Timeline), ISOMITTED(APR)),
    //  A blank cell is not an omitted argument, so test for both
        Method,         IF(OR(ISOMITTED(Convention), NOT(ISNUMBER(Convention))), 3, Convention),
        EndFlag,        IF(ISLOGICAL(EndDates), EndDates, IF(ISNUMBER(EndDates), EndDates <> 0, FALSE)),
    //  One timeline laid along a row, text dates read, and a column of rates laid the same way
        Row,            IF(ROWS(Timeline) = 1, Timeline, TRANSPOSE(Timeline)),
        Dates,          IF(ISNUMBER(Row), Row, DATEVALUE(Row)),
        Rate,           IF(ROWS(APR) = 1, APR, TRANSPOSE(APR)),
        Count,          COLUMNS(Dates),
    //  The period length is read off the first gap for the first period and the last gap
    //  for the last, in whole months from 28 days up and in days below that. A month-end
    //  date steps back through EOMONTH, because EDATE(28 Feb, -1) is 28 Jan, not 31 Jan.
        FirstGap,       INDEX(Dates, 1, 2) - INDEX(Dates, 1, 1),
        LastGap,        INDEX(Dates, 1, Count) - INDEX(Dates, 1, Count - 1),
        FirstDate,      INDEX(Dates, 1, 1),
        FirstMonths,    ROUND(FirstGap / 30.5, 0),
        FirstStart,     IF(FirstGap < 28,
                            FirstDate - FirstGap,
                            IF(DAY(FirstDate + 1) = 1,
                                EOMONTH(FirstDate, -FirstMonths),
                                EDATE(FirstDate, -FirstMonths))) + 1,
        LastDate,       INDEX(Dates, 1, Count),
        LastMonths,     ROUND(LastGap / 30.5, 0),
        LastEnd,        IF(LastGap < 28,
                            LastDate + LastGap,
                            IF(DAY(LastDate + 1) = 1,
                                EOMONTH(LastDate, LastMonths),
                                EDATE(LastDate, LastMonths))) - 1,
    //  Every period runs from its start to the day before the next start
        Starts,         IF(EndFlag, HSTACK(FirstStart, DROP(Dates, , -1) + 1), Dates),
        Ends,           IF(EndFlag, Dates, HSTACK(DROP(Dates, , 1) - 1, LastEnd)),
        Next,           Ends + 1,
    //  30/360 on the European rule: a 31st counts as the 30th at either end
        Days,           IF(Method = 1,
                            360 * (YEAR(Next) - YEAR(Starts)) + 30 * (MONTH(Next) - MONTH(Starts))
                                + IF(DAY(Next) > 30, 30, DAY(Next)) - IF(DAY(Starts) > 30, 30, DAY(Starts)),
                            Next - Starts),
    //  Actual/Actual counts partial first and last years, plus any whole years between.
    //  IF rather than MIN, because MIN would collapse the row to one value.
        YearOne,        YEAR(Starts),
        Boundary,       DATE(YearOne + 1, 1, 1),
        Split,          IF(Next < Boundary, Next, Boundary),
        DaysOne,        IF(DAY(DATE(YearOne, 2, 29)) = 29, 366, 365),
        YearLast,       YEAR(Next),
        LastYearStart,  DATE(YearLast, 1, 1),
        DaysLast,       DATE(YearLast + 1, 1, 1) - LastYearStart,
        Fraction,       SWITCH(Method,
                            1, Days / 360,
                            2, Days / 360,
                            4, (Split - Starts) / DaysOne + IF(YearLast > YearOne,
                                YearLast - YearOne - 1 + (Next - LastYearStart) / DaysLast, 0),
                            Days / 365),
        Rates,          Rate * Fraction,
        Result,         IF(OR(Count < 2, NOT(OR(Method = {1,2,3,4}))), #VALUE!, Rates),
    //  Return Result or Help
        CHOOSE( Help? + 1, Result, Help)
    )
);

/*  FUNCTION NAME:  ROEλ
    DESCRIPTION:*//**Return on equity (ROE) ratio is a measure of financial performance (Net Income/Average Shareholders' Equity)*/
ROEλ = LAMBDA(
//  Parameter Declarations
        [NetIncome],
        [ShareholdersEquity],
        [OpeningEquity],
//  Help
    LET(Help,   TRIM(TEXTSPLIT(
                        "FUNCTION:          →ROEλ( NetIncome, ShareholdersEquity, [OpeningEquity])¶" &
                        "DESCRIPTION:       →Return on equity ratio is considered a gauge of a corporation's ¶" &
                                           "→profitability and how efficient it is in generating profits¶" &
                                           "→(Net Income/ Average Shareholders' Equity)¶" &
                        "WEBPAGE:           →https://www.investopedia.com/terms/r/returnonequity.asp ¶" &
                        "VERSION:           →6 Sep 2026¶" &
                        "PARAMETERS:→¶" &
                        "NetIncome          →(Required) Net Income¶" &
                        "ShareholdersEquity →(Required) Shareholders' Equity at the period close. Give the average of open and close here when OpeningEquity is left out ¶" &
                        "OpeningEquity      →(Optional) Shareholders' Equity at the period open. When given, the ratio divides by the average of OpeningEquity and ShareholdersEquity ¶" &
                        "EXAMPLES:          →Formula (oz is assumed to be the module's name)¶" &
                        "23.6213%           →=oz.ROEλ( 59.972, 256.144, 251.635 )¶" &
                        "23.6213%           →=oz.ROEλ( 59.972, AVERAGE(251.635, 256.144) )",
                    "→", "¶" )
                ),
    //  Check inputs - Omitted required arguments (do not include optional arguments)
        Help?,          OR( ISOMITTED( NetIncome),
                            ISOMITTED( ShareholdersEquity)
                        ),
    //  A blank OpeningEquity cell is not an omitted argument, so test for both: either
    //  way the closing equity stands in for it and the ratio is what it was before
        Opening,        IF( ISOMITTED( OpeningEquity),
                            ShareholdersEquity,
                            IF( ISNUMBER( OpeningEquity), OpeningEquity, ShareholdersEquity)
                        ),
    //  Procedure
        Result,         NetIncome / ((ShareholdersEquity + Opening) / 2),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)        
    )
);

/*  FUNCTION NAME:      IsBetweenUλ
    DESCRIPTION:*//**Determine if a value is between a lower and upper limit*/

//  Parameter Declarations
    [Value],
    [Low],
    [Hi],
    [Inclusive],
//  Help
    LET(Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →IsBetweenUλ( Value, Low, Hi, [Inclusive])¶" &
                            "DESCRIPTION:   →Determine if a value is between a lower and upper limit.¶" & 
                            "PARAMETERS:    →¶" & 
                            "Value          →(Required) A value or array of values to compare.¶" & 
                            "Low            →(Required) The lower limit that the value must be greater than¶" &
                            "Hi             →(Required) The higher limit that the value must be less than¶" &
                            "Inclusive      →(Optional) If set to TRUE (default) the value can be equal to Low and/or Hi¶" &
                            "→¶" &
                            "EXAMPLES:      →¶" & 
                            "Result         →Formula¶" &
                            "FALSE          →=IsBetweenUλ(1, 2, 4)¶" & 
                            "TRUE           →=IsBetweenUλ(2, 2, 4)¶" & 
                            "FALSE          →=IsBetweenUλ(2, 2, 4, FALSE)", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Set optional arguments defaults
        Inclusive,      IF( ISLOGICAL( Inclusive), Inclusive, TRUE),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Value),
                            ISOMITTED( Low),
                            ISOMITTED( Hi)
                        ),
    //  Procedure
        Result,         IF( Inclusive, 
                            (Value >= Low) * (Value <= Hi) = 1, 
                            (Value > Low) * (Value < Hi) = 1
                        ),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )
);

        Rows,           IF( ISNUMBER( Rows), Rows, 1),
        Columns,        IF( ISNUMBER( Columns), Columns, 1),
    //  Procedure
        Result,         OFFSET( Range, 0, 0, Rows, Columns),

/*  Module Contains Ozzit Compliant LAMBDAs for debt sculpting and amortisation.
    Every schedule solves its closing balances with SCAN, one period after another, so
    nothing here recurses and the module is held in the Advanced Formula Environment
    store with the other five. */

/*  FUNCTION NAME:  AmortiseBλ
    DESCRIPTION:*//**Creates an amortisation schedule for a debt/loan across a model's periods*/
AmortiseBλ = LAMBDA(
//  Parameter Declarations
    [Principal],
    [APR],
    [Term],
    [ModelPeriods],
    [StartPeriod],
    [MonthsPerPeriod],
//  Help
    LET(Help,       TRIM(TEXTSPLIT(
                        "FUNCTION:      →AmortiseBλ(Principal, APR, Term, ModelPeriods, [StartPeriod], [MonthsPerPeriod])¶" &
                        "DESCRIPTION:   →Creates an amortisation schedule for a debt/loan across a model's periods.¶" &
                        "ROW CONTENTS:  →Opening Balance¶" &
                        "               →Interest Portion¶" &
                        "               →Repayment¶" &
                        "               →Closing Balance¶" &
                        "               →Principal Portion¶" &
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                        "VERSION:       →20 Aug 2026¶" &
                        "PARAMETERS:    →¶" &
                        "Principal      →(Required) Principal opening balance.¶" &
                        "APR            →(Required) Annual percentage rate.¶" &
                        "Term           →(Required) Number of periods to pay off the debt.¶" &
                        "ModelPeriods   →(Required) Number of model periods.¶" &
                        "StartPeriod    →(Optional) Period when the debt is issued. Defaults to 1.¶" &
                        "MonthsPerPeriod→(Optional) Number of months in each period. Defaults to 1.¶" &
                        "EXAMPLES:      →¶" &
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                        "5 rows, 60 cols→=oz.AmortiseBλ(10000, 3%, 48, 60)",
                        "→", "¶"
                        )
                    ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Principal),
                            ISOMITTED( APR),
                            ISOMITTED( Term),
                            ISOMITTED( ModelPeriods)
                        ),
    //  Check inputs - Errors in argument values
        ErrMsgs,        {"Principal must be a single number providing the opening debt amount.";
                         "APR must be a single number providing annual percentage rate for interest.";
                         "Term must be a positive whole number of periods to pay off the debt";
                         "Model Periods must be a positive whole number of periods in this model."},
    //  Term reaches PMT( Rate, Term, Opening) and ModelPeriods reaches SEQUENCE:
    //  a zero, negative or fractional value returns #NUM! rather than this message.
        Errors,         VSTACK( NOT( ISNUMBER( Principal)),
                                NOT( ISNUMBER( APR)),
                                NOT( AND( ISNUMBER( Term), Term > 0, Term = INT( Term))),
                                NOT( AND( ISNUMBER( ModelPeriods), ModelPeriods > 0,
                                          ModelPeriods = INT( ModelPeriods)))
                        ),
        Messages,       FILTER( ErrMsgs, Errors, ""),
    //  Check inputs - Set optional arguments' defaults
        StartPeriod,    IF( ISNUMBER( StartPeriod), StartPeriod, 1),
        MonthsPerPeriod, IF( ISNUMBER( MonthsPerPeriod), MonthsPerPeriod, 1),
    //  Procedure - solve the closing balance period by period, then read the rows off it.
    //  Before the start period the balance is carried untouched; from it, each period pays
    //  the level payment for the term that remains, capped at the balance plus interest.
        Rate,           APR / 12 * MonthsPerPeriod,
        Periods,        SEQUENCE( 1, ModelPeriods),
        Closings,       SCAN( Principal, Periods,
                            LAMBDA( Carried, Period,
                                LET(Remaining,  IF( Period < StartPeriod, 0, MAX( Term - Period + StartPeriod, 0)),
                                    Opening,    IF( Remaining = 0, 0, Carried),
                                    Interest,   Opening * Rate,
                                    Payment,    IF( Remaining = 0, 0, MIN( -PMT( Rate, Remaining, Opening), Opening + Interest)),
                                    IF( Period < StartPeriod, Carried, Opening + Interest - Payment)
                                )
                            )
                        ),
        Carried,        IF( ModelPeriods = 1, Principal, HSTACK( Principal, DROP( Closings, , -1))),
        Remaining,      IF( Periods < StartPeriod, 0, IF( Term - Periods + StartPeriod > 0, Term - Periods + StartPeriod, 0)),
        Opening,        IF( Remaining = 0, 0, Carried),
        Interest,       Opening * Rate,
        Level,          IF( Remaining = 0, 0, -PMT( Rate, Remaining, Opening)),
        Payment,        IF( Level < Opening + Interest, Level, Opening + Interest),
        Closing,        Opening + Interest - Payment,
        Portion,        Payment - Interest,
        Result,         VSTACK( Opening, Interest, Payment, Closing, Portion),
    //  Return Result, Help or the messages
        Return,         IF( Help?, 2, IF( OR( Errors), 3, 1)),
        CHOOSE( Return, Result, Help, Messages)
    )
);



/*  FUNCTION NAME:  DebtSculptFixedλ
    DESCRIPTION:*//**Creates a debt sculpting schedule where APR and DSCR are fixed for all periods*/
DebtSculptFixedλ = LAMBDA(
//  Parameter Declarations
    [OpeningBalance],
    [Debt],
    [CFADS],
    [DSCR],
    [APR],
    [MonthsPerPeriod],
//  Help
    LET(Help,       TRIM(TEXTSPLIT(
                        "FUNCTION:          →DebtSculptFixedλ([OpeningBalance], Debt, CFADS, DSCR, APR, [MonthsPerPeriod])¶" &
                        "DESCRIPTION:       →Creates a Debt Sculpting schedule where APR and DSCR are fixed for all periods¶" &
                        "SUGGESTED ROW IDS: →Debt opening balance¶" &
                        "                   →Interest on Principal Balance¶" &
                        "                   →Debt service (interest and principal)¶" &
                        "                   →Debt closing balance¶" &
                        "WEBPAGE:           →https://github.com/ryanduguid/Ozzit¶" &
                        "VERSION:           →10 Sep 2026¶" &
                        "PARAMETERS:        →¶" &
                        "OpeningBalance     →(Optional) Model's debt on opening. Defaults to 0¶" &
                        "Debt               →(Required) One row of new debt, at least two periods. Every period input must have this same width¶" &
                        "CFADS              →(Required) A row of Cash Flow Available for Debt Service in each period¶" &
                        "DSCR               →(Required) Debt Service Coverage Ratio, one number for all periods¶" &
                        "APR                →(Required) Interest's Annual Percentage Rate, one number for all periods¶" &
                        "MonthsPerPeriod    →(Optional) Defaults to one month per period.¶" &
                        "EXAMPLES:          →¶" &
                        "Result             →Formula (oz is assumed to be the module's name)¶" &
                        "4 rows, 3 cols     →=oz.DebtSculptFixedλ(, {1000,0,0}, {300,300,300}, 1.2, 6%, 12)",
                        "→", "¶"
                        )
                    ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Debt),
                            ISOMITTED( CFADS),
                            ISOMITTED( DSCR),
                            ISOMITTED( APR)
                        ),
    //  Check inputs - Errors in argument values
        ErrMsgs,        {"Debt must be one row of new debt with at least two periods";
                         "CFADS must be one row with the same number of periods as Debt";
                         "DSCR must be a single positive number holding debt service coverage ratio for all periods";
                         "APR must be a single number holding interest's annual percentage rate for all periods"},
    //  DSCR divides Cash below, so zero returns #DIV/0! rather than this message.
        Errors,         VSTACK( OR( ROWS( Debt) <> 1, COLUMNS( Debt) <= 1),
                                OR( ROWS( CFADS) <> 1, COLUMNS( CFADS) <> COLUMNS( Debt)),
                                NOT( AND( ISNUMBER( DSCR), ROWS( DSCR) = 1, COLUMNS( DSCR) = 1,
                                          DSCR > 0)),
                                NOT( AND( ISNUMBER( APR), ROWS( APR) = 1, COLUMNS( APR) = 1))
                        ),
        Messages,       FILTER( ErrMsgs, Errors, ""),
    //  Check inputs - Set optional arguments' defaults
        OpeningBalance, IF( ISNUMBER( OpeningBalance), OpeningBalance, 0),
        MonthsPerPeriod, IF( ISNUMBER( MonthsPerPeriod), MonthsPerPeriod, 1),
    //  Procedure - solve the closing balance period by period, then read the rows off it
    //  Negative CFADS means no cash for debt service, so the payment is floored at
    //  zero. Without the floor it went negative, which raised the closing balance and
    //  reported the negated amount as positive debt service. The rows below floor with
    //  IF rather than MAX: MAX would aggregate the whole row to one number. They floor
    //  the amount owed too, so a negative opening balance repays nothing in the rows
    //  exactly as it does in the SCAN, and each closing balance opens the next period.
        Rate,           APR / 12 * MonthsPerPeriod,
        Periods,        SEQUENCE( 1, COLUMNS( Debt)),
        NewDebt,        INDEX( Debt, 1, Periods),
        Cash,           INDEX( CFADS, 1, Periods),
        Closings,       SCAN( OpeningBalance, Periods,
                            LAMBDA( Carried, Period,
                                LET(Opening,    Carried + INDEX( NewDebt, 1, Period),
                                    Interest,   Opening * Rate,
                                    Payment,    MAX( 0, MIN( INDEX( Cash, 1, Period) / DSCR, Opening + Interest)),
                                    Opening + Interest - Payment
                                )
                            )
                        ),
        Carried,        IF( COLUMNS( Debt) = 1, OpeningBalance, HSTACK( OpeningBalance, DROP( Closings, , -1))),
        Opening,        Carried + NewDebt,
        Interest,       Opening * Rate,
        Service,        IF( Cash / DSCR < 0, 0, Cash / DSCR),
        Payment,        IF( Opening + Interest < 0, 0, IF( Service < Opening + Interest, Service, Opening + Interest)),
        Closing,        Opening + Interest - Payment,
        Result,         VSTACK( Opening, Interest, -Payment, Closing),
    //  Return Result, Help or the messages
        Return,         IF( Help?, 2, IF( OR( Errors), 3, 1)),
        CHOOSE( Return, Result, Help, Messages)
    )
);



/*  FUNCTION NAME:  DebtSculptVariableLRVλ
    DESCRIPTION:*//**Creates a debt sculpting schedule where APR and DSCR are set for each period, with interest charged on the average balance*/
DebtSculptVariableLRVλ = LAMBDA(
//  Parameter Declarations
    [OpeningBalance],
    [Debt],
    [CFADS],
    [DSCR],
    [APR],
    [MonthsPerPeriod],
    [PeriodRates],
//  Help
    LET(Help,       TRIM(TEXTSPLIT(
                        "FUNCTION:          →DebtSculptVariableLRVλ([OpeningBalance], Debt, CFADS, DSCR, APR, [MonthsPerPeriod], [PeriodRates])¶" &
                        "DESCRIPTION:       →Creates a Debt Sculpting schedule where APR and DSCR are set for each period.¶" &
                        "                   →Interest is charged on the average balance over the period, the method¶" &
                        "                   →presented by Lance Rubin, and the principal repaid is what the cash¶" &
                        "                   →available for debt service leaves after that interest.¶" &
                        "SUGGESTED ROW IDS: →Debt opening balance¶" &
                        "                   →Interest on Principal Balance¶" &
                        "                   →Principal repayments¶" &
                        "                   →Debt closing balance¶" &
                        "WEBPAGE:           →https://github.com/ryanduguid/Ozzit¶" &
                        "VERSION:           →10 Sep 2026¶" &
                        "PARAMETERS:        →¶" &
                        "OpeningBalance     →(Optional) Model's debt on opening. Defaults to 0¶" &
                        "Debt               →(Required) One row of new debt, at least two periods. Every period input must have this same width¶" &
                        "CFADS              →(Required) A row of Cash Flow Available for Debt Service in each period¶" &
                        "DSCR               →(Required) A row of Debt Service Coverage Ratios, one per period¶" &
                        "APR                →(Required unless PeriodRates is given) A row of Interest's Annual Percentage Rates, one per period. Each period charges APR / 12 * MonthsPerPeriod¶" &
                        "MonthsPerPeriod    →(Optional) Defaults to one month per period. Ignored when PeriodRates is given¶" &
                        "PeriodRates        →(Optional) A row of interest rates per period, one per period, such as DayCountRateλ() returns for a day count convention. When given, interest uses these rates and APR and MonthsPerPeriod are ignored. A blank cell or range counts as not given¶" &
                        "EXAMPLES:          →¶" &
                        "Result             →Formula (oz is assumed to be the module's name)¶" &
                        "4 rows, 3 cols     →=oz.DebtSculptVariableLRVλ(, {1000,0,0}, {300,300,300}, {1.2,1.2,1.2}, {0.06,0.06,0.06}, 12)",
                        "→", "¶"
                        )
                    ),
    //  A blank PeriodRates cell or range is not a row of rates, so test for both
        UseRates,       AND( NOT( ISOMITTED( PeriodRates)), COUNT( PeriodRates) > 0),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Debt),
                            ISOMITTED( CFADS),
                            ISOMITTED( DSCR),
                            AND( ISOMITTED( APR), NOT( UseRates))
                        ),
    //  Check inputs - Errors in argument values
        ErrMsgs,        {"Debt must be one row of new debt with at least two periods";
                         "CFADS must be one row with the same number of periods as Debt";
                         "DSCR must be one row of positive numbers with the same number of periods as Debt";
                         "APR or PeriodRates must be one row with the same number of periods as Debt"},
    //  Every DSCR element divides its period's cash below, so one zero returns
    //  #DIV/0! for the whole schedule rather than this message.
        Errors,         VSTACK( OR( ROWS( Debt) <> 1, COLUMNS( Debt) <= 1),
                                OR( ROWS( CFADS) <> 1, COLUMNS( CFADS) <> COLUMNS( Debt)),
                                OR( ROWS( DSCR) <> 1, COLUMNS( DSCR) <> COLUMNS( Debt),
                                    NOT( AND( ISNUMBER( DSCR))), OR( DSCR <= 0)),
                                IF( UseRates,
                                    OR( ROWS( PeriodRates) <> 1, COLUMNS( PeriodRates) <> COLUMNS( Debt)),
                                    OR( ROWS( APR) <> 1, COLUMNS( APR) <> COLUMNS( Debt)))
                        ),
        Messages,       FILTER( ErrMsgs, Errors, ""),
    //  Check inputs - Set optional arguments' defaults
        OpeningBalance, IF( ISNUMBER( OpeningBalance), OpeningBalance, 0),
        MonthsPerPeriod, IF( ISNUMBER( MonthsPerPeriod), MonthsPerPeriod, 1),
    //  Procedure - solve the closing balance period by period, then read the rows off it
    //  A payment below zero is deliberate here: this function pays interest out of the
    //  period's cash and repays principal, so an unpaid period capitalises its interest
    //  through a negative payment. tools/excel_selftest.ps1 pins it as "no cash
    //  capitalises interest once". Do not floor it without changing that assertion.
        Periods,        SEQUENCE( 1, COLUMNS( Debt)),
    //  A row of rates per period, from a day count convention, replaces the flat twelfth of APR
        Rates,          IF( UseRates,
                            INDEX( PeriodRates, 1, Periods),
                            INDEX( APR, 1, Periods) / 12 * MonthsPerPeriod
                        ),
        NewDebt,        INDEX( Debt, 1, Periods),
        Cash,           INDEX( CFADS, 1, Periods),
        Cover,          INDEX( DSCR, 1, Periods),
        Closings,       SCAN( OpeningBalance, Periods,
                            LAMBDA( Carried, Period,
                                LET(Opening,    Carried + INDEX( NewDebt, 1, Period),
                                    Interest,   InterestLRVλ( INDEX( Cash, 1, Period), INDEX( Cover, 1, Period), Opening, INDEX( Rates, 1, Period)),
                                    Payment,    MIN( INDEX( Cash, 1, Period) / INDEX( Cover, 1, Period) - Interest, Opening),
                                    Opening - Payment
                                )
                            )
                        ),
        Carried,        IF( COLUMNS( Debt) = 1, OpeningBalance, HSTACK( OpeningBalance, DROP( Closings, , -1))),
        Opening,        Carried + NewDebt,
        Interest,       MAP( Periods,
                            LAMBDA( Period,
                                InterestLRVλ( INDEX( Cash, 1, Period), INDEX( Cover, 1, Period), INDEX( Opening, 1, Period), INDEX( Rates, 1, Period))
                            )
                        ),
        Available,      Cash / Cover - Interest,
        Payment,        IF( Available < Opening, Available, Opening),
        Closing,        Opening - Payment,
        Result,         VSTACK( Opening, Interest, -Payment, Closing),
    //  Return Result, Help or the messages
        Return,         IF( Help?, 2, IF( OR( Errors), 3, 1)),
        CHOOSE( Return, Result, Help, Messages)
    )
);

/*  FUNCTION NAME:  InterestLRVλ
    DESCRIPTION:*//**Calculates debt sculpting interest on the average balance over the period, the method presented by Lance Rubin*/
InterestLRVλ = LAMBDA(
//  Parameter Declarations
    [CFADS],
    [DSCR],
    [Principal],
    [InterestRate],
//  Help
    LET(Help,       TRIM(TEXTSPLIT(
                        "FUNCTION:      →InterestLRVλ(CFADS, DSCR, Principal, InterestRate)¶" &
                        "DESCRIPTION:   →Calculates debt sculpting interest using the method presented by Lance Rubin.¶" &
                                       "→Interest is charged on the average balance over the period, with the¶" &
                                       "→principal repaid capped at the principal outstanding. The average balance¶" &
                                       "→depends on the interest and the interest on the average balance, which¶" &
                                       "→is solved in closed form rather than by iteration.¶" &
                        "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                        "VERSION:       →20 Aug 2026¶" &
                        "PARAMETERS:    →¶" &
                        "CFADS          →(Required) Cash Flow Available for Debt Service for this period¶" &
                        "DSCR           →(Required) Debt Service Coverage Ratio for this period¶" &
                        "Principal      →(Required) Principal balance for this period¶" &
                        "InterestRate   →(Required) Interest rate for this period (eg. APR/Days in this year * Days in this period)¶" &
                        "EXAMPLES:      →¶" &
                        "Result         →Formula (oz is assumed to be the module's name)¶" &
                        "222.90         →=oz.InterestLRVλ( 6666.37, 3.50, 90000, 3%/12)",
                        "→", "¶"
                        )
                    ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( CFADS),
                            ISOMITTED( DSCR),
                            ISOMITTED( Principal),
                            ISOMITTED( InterestRate)
                        ),
    //  Check inputs - Errors in argument values
        ErrMsgs,        {"CFADS must be a single number holding cash available for debt service for this period";
                         "DSCR must be a single positive number holding debt service coverage ratio for this period";
                         "Principal must be a single number holding this period's opening debt balance";
                         "InterestRate must be a single number holding this period's interest rate (APR/Days in this year * Days in this period)"},
        Errors,         VSTACK( NOT( AND( ISNUMBER( CFADS), COLUMNS( CFADS) = 1)),
                                NOT( AND( ISNUMBER( DSCR), COLUMNS( DSCR) = 1, DSCR > 0)),
                                NOT( AND( ISNUMBER( Principal), COLUMNS( Principal) = 1)),
                                NOT( AND( ISNUMBER( InterestRate), COLUMNS( InterestRate) = 1))
                        ),
        Messages,       FILTER( ErrMsgs, Errors, ""),
    //  Procedure - the fixed point of Interest = (Principal - Payment / 2) * Rate with
    //  Payment = MIN( DebtService - Interest, Principal), solved in closed form. Where the
    //  cash would repay more than the principal, the payment is the principal and half a
    //  period's interest is charged on it.
        DebtService,    CFADS / DSCR,
        Capped?,        DebtService >= Principal * ( 1 + InterestRate / 2),
        Interest,       IF( Capped?,
                            Principal * InterestRate / 2,
                            InterestRate * ( 2 * Principal - DebtService) / ( 2 - InterestRate)
                        ),
        Result,         MAX( Interest, 0),
    //  Return Result, Help or the messages
        Return,         IF( Help?, 2, IF( OR( Errors), 3, 1)),
        CHOOSE( Return, Result, Help, Messages)
    )
);
