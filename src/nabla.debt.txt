Amortiseλ =
LAMBDA(Principal,APR,Term,ModelPeriods,StartPeriod,MonthsPerPeriod,DoNotUse, LET(Help, TRIM(TEXTSPLIT("DESCRIPTION:   →Creates an amortisation schedule for a debt/loan¶" & "ROW CONTENTS:  →Opening Balance¶" & "               →Interest Portion¶" & "               →Repayment¶" & "               →Closing Balance¶" & "               →Principal Portion¶" & "VERSION:       →18 Aug 2026¶" & "PARAMETERS:→¶" & "Principal      →(Required) Principal opening balance.¶" & "APR            →(Required) APR¶" & "Term           →(Required) Number of periods to pay off debt¶" & "ModelPeriods   →(Required) Number of model periods¶" & "StartPeriod    →(Optional) Period when debt is issued. Defaults to 1.¶" & "MonthsPerPeriod→(Optional) Number of months in each period. Defaults to 1.¶" & "DoNotUse       →DO NOT USE - Internal use only (Period Counter)¶" & "→¶" & "EXAMPLES:      →Amortiseλ(10000, 3%, 48, 60)", "→", "¶")), ErrMsgs, {"Principal must be a single number providing the opening debt amount.";"APR must be a single number providing annual percentage rate for interest.";"Term must be a single number providing the number of periods to pay off the debt";"Model Periods must be a single number providing the number of periods in this model."}, Principal, IF(ISNUMBER(Principal), Principal, #VALUE!), APR, IF(ISNUMBER(APR), APR, #VALUE!), Term, IF(ISNUMBER(Term), Term, #VALUE!), ModelPeriods, IF(ISNUMBER(ModelPeriods), ModelPeriods, #VALUE!), StartPeriod, IF(ISNUMBER(StartPeriod), StartPeriod, 1), MonthsPerPeriod, IF(ISNUMBER(MonthsPerPeriod), MonthsPerPeriod, 1), Errors, VSTACK(ISERROR(Principal), ISERROR(APR), ISERROR(Term), ISERROR(ModelPeriods)), Messages, FILTER(ErrMsgs, Errors, ""), Period, IF(DoNotUse = 0, 1, DoNotUse), Rate, APR / 12 * MonthsPerPeriod, RemainingTerm, IF(Period < StartPeriod, 0, MAX(Term - Period + StartPeriod, 0)), Payment, IF(RemainingTerm = 0, 0, -PMT(Rate, RemainingTerm, Principal)), OpeningBalance, IF(RemainingTerm = 0, 0, Principal), InterestPortion, OpeningBalance * Rate, NewPayment, IF(Period < StartPeriod, 0, MIN(Payment, OpeningBalance + InterestPortion)), PrincipalPortion, NewPayment - InterestPortion, ClosingBalance, OpeningBalance + InterestPortion - NewPayment, NewColumn, VSTACK(OpeningBalance, InterestPortion, NewPayment, ClosingBalance, PrincipalPortion), Result, IF(Period = ModelPeriods, NewColumn, HSTACK(NewColumn, [0]!nabla.debt.Amortiseλ(IF(Period < StartPeriod, Principal, ClosingBalance), APR, Term, ModelPeriods, StartPeriod, MonthsPerPeriod, Period + 1))), Error, IF(AND(Errors), 2, IF(OR(Errors), 3, IF(OR(ISERROR(Result)), 2, 1))), CHOOSE(Error, Result, Help, Messages)));
