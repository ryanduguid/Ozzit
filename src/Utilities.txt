/*  FUNCTION NAME:  CountCUλ
    DESCRIPTION:*//**Count occurrences of character(s) in a string*/
CountCUλ = LAMBDA(
//  Parameter Declarations
        [Text],
        [Characters],
//  Help
    LET(Help,       TRIM(TEXTSPLIT( 
                            "FUNCTION:      →CountCUλ( Text, Characters)¶" &
                            "DESCRIPTION:   →Count how many of one or more characters are in a string.¶" & 
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit¶" &
                            "VERSION:       →20 Aug 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Text           →(Required) The string to interrogate¶" & 
                            "Characters     →(Required) A single character, or CSV of characters to find in Text¶" & 
                                           "→NOTE! Case sensitive¶" & 
                            "EXAMPLES:      →Formula (oz is assumed to be the module's name)¶" &
                                           "→=oz.CountCUλ( ""Mary had a little lamb"", ""a,b"")", 
                        "→", "¶" )
                    ),
    //  Check inputs - Omitted required arguments
       Help?,           OR( ISOMITTED( Text),
                            ISOMITTED( Characters)
                        ),
    //  Set defaults
        Characters,     TEXTSPLIT( Characters, "," ),
    //  Procedure
        Result,         LEN(Text) - SUM(LEN(TEXTSPLIT( Text, Characters))),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)        
    )
);

/*  FUNCTION NAME:      IsBetweenUλ
    DESCRIPTION:*//**Determine if a value is between a lower and upper limit*/
IsBetweenUλ = LAMBDA(
//  Parameter Declarations
    [Value],
    [Low],
    [Hi],
    [Inclusive],
//  Help
    LET(Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →IsBetweenUλ( Value, Low, Hi, [Inclusive])¶" &
                            "DESCRIPTION:   →Determine if a value is between a lower and upper limit.¶" & 
                            "PARAMETERS:    →¶" & 
                            "Value          →(Required) A value or array of values to compare.¶" & 
                            "Low            →(Required) The lower limit that the value must be greater than¶" &
                            "Hi             →(Required) The higher limit that the value must be less than¶" &
                            "Inclusive      →(Optional) If set to TRUE (default) the value can be equal to Low and/or Hi¶" &
                            "→¶" &
                            "EXAMPLES:      →¶" & 
                            "Result         →Formula¶" &
                            "FALSE          →=IsBetweenUλ(1, 2, 4)¶" & 
                            "TRUE           →=IsBetweenUλ(2, 2, 4)¶" & 
                            "FALSE          →=IsBetweenUλ(2, 2, 4, FALSE)", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Set optional arguments defaults
        Inclusive,      IF( ISLOGICAL( Inclusive), Inclusive, TRUE),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Value),
                            ISOMITTED( Low),
                            ISOMITTED( Hi)
                        ),
    //  Procedure
        Result,         IF( Inclusive, 
                            (Value >= Low) * (Value <= Hi) = 1, 
                            (Value > Low) * (Value < Hi) = 1
                        ),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )
);



/*  FUNCTION NAME:      IsInListUλ
    DESCRIPTION:*//**Determine whether a value is one of a list of items*/
IsInListUλ = LAMBDA(
//  Parameter Declarations
    [Value],
    [List],
//  Help
    LET(Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →IsInListUλ( Value, List)¶" &
                            "DESCRIPTION:   →Determine if a value is one of a list of items.¶" & 
                            "PARAMETERS:    →¶" & 
                            "Value          →(Required) A value or array of values to compare.¶" & 
                            "List           →(Required) A list of values to check: a row, a column or a grid.¶" &
                            "NOTE!          →The comparison is exact and case-insensitive; * and ? are ordinary characters.¶" &
                            "→¶" &
                            "EXAMPLES:      →¶" & 
                            "Result         →Formula¶" &
                            "FALSE          →=IsInListUλ(1, {2,3,4})¶" & 
                            "TRUE           →=IsInListUλ(1, {1,3,5})¶" & 
                            "→", "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Value),
                            ISOMITTED( List)
                        ),
    //  Procedure
    //  Flatten the list first: MATCH needs one dimension and treats * and ? as wildcards
        Result,         ISNUMBER( XMATCH( Value, TOCOL( List), 0)),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )
);



/*  FUNCTION NAME:  RangeToDAUλ
    DESCRIPTION:*//**Convert a range reference to a dynamic array reference*/
RangeToDAUλ = LAMBDA(
//  Parameter Declarations
    [Range],
    [Rows],
    [Columns],
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:          →RangeToDAUλ(Range, [Rows], [Columns])¶" &
                            "DESCRIPTION:       →Convert a range reference to a dynamic array reference.¶" & 
                            "SEE ALSO:      →Excel 365 now has TRIMRANGE and trim references (.:.) for this.¶" &
                            "WEBSITE:           →https://github.com/ryanduguid/Ozzit¶" &
                            "PARAMETERS:        →¶" & 
                            "Range              →(Required) A reference to the range to be converted.¶" &
                            "Rows               →(Optional) Number of rows for the dynamic array. Defaults to 1.¶" &
                            "Columns            →(Optional) Number of columns for the dynamic array. Defaults to 1.¶" &
                            "→¶" &
                            "EXAMPLES:          →=RangeToDAUλ(A1, , ModelPeriodCount)",
                            "→", "¶"
                            )
                        ),  
    //  Check inputs - Omitted required arguments
        Help?,          ISOMITTED( Range),
    //  Check inputs - Set optional arguments' defaults
        Rows,           IF( ISNUMBER( Rows), Rows, 1),
        Columns,        IF( ISNUMBER( Columns), Columns, 1),
    //  Procedure
        Result,         OFFSET( Range, 0, 0, Rows, Columns), 
    //  Return Result or help
        CHOOSE(Help? + 1, Result, Help)
    )
);
