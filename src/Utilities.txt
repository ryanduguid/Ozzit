/*  FUNCTION NAME:  CountCUλ
    DESCRIPTION:*//**Count occurrences of character(s) in a string*/
CountCUλ = LAMBDA(
//  Parameter Declarations
        [Text],
        [Characters],
//  Help
    LET(Help,       TRIM(TEXTSPLIT( 
                            "FUNCTION:      →CountCUλ( Text, Characters)¶" &
                            "DESCRIPTION:   →Count how many of one or more characters are in a string.¶" & 
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit/blob/main/docs/function-guide.md¶" &
                            "VERSION:       →14 Sep 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Text           →(Required) The string to interrogate. Empty text returns 0¶" & 
                            "Characters     →(Required) A single character, or CSV of characters to find in Text¶" & 
                                           "→NOTE! Case sensitive. A single character is taken as itself, so a comma works¶" & 
                            "EXAMPLES:      →Formula (oz is assumed to be the module's name)¶" &
                                           "→=oz.CountCUλ( ""Mary had a little lamb"", ""a,b"")", 
                        "→", "¶" )
                    ),
    //  Check inputs - Omitted required arguments
       Help?,           OR( ISOMITTED( Text),
                            ISOMITTED( Characters)
                        ),
    //  Set defaults. One character is taken as itself, so a comma can be asked for;
    //  2 or more are read as the comma separated list the help describes.
        Characters,     IF( LEN( Characters) = 1, Characters, TEXTSPLIT( Characters, "," )),
    //  Procedure. Empty text holds none of them, and TEXTSPLIT has nothing to split.
        Result,         IF( LEN( Text) = 0, 0,
                            LEN( Text) - SUM( LEN( TEXTSPLIT( Text, Characters)))),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)        
    )
);

/*  FUNCTION NAME:      IsBetweenUλ
    DESCRIPTION:*//**Determine if a value is between a lower and upper limit*/
IsBetweenUλ = LAMBDA(
//  Parameter Declarations
    [Value],
    [Low],
    [Hi],
    [Inclusive],
//  Help
    LET(Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →IsBetweenUλ( Value, Low, Hi, [Inclusive])¶" &
                            "DESCRIPTION:   →Determine if a value is between a lower and upper limit.¶" & 
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit/blob/main/docs/function-guide.md¶" &
                            "VERSION:       →24 Oct 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Value          →(Required) A value or array of values to compare.¶" & 
                            "Low            →(Required) The lower limit that the value must be greater than¶" &
                            "Hi             →(Required) The higher limit that the value must be less than¶" &
                            "Inclusive      →(Optional) If set to TRUE (default) the value can be equal to Low and/or Hi¶" &
                            "→¶" &
                            "EXAMPLES:      →¶" & 
                            "Result         →Formula¶" &
                            "FALSE          →=IsBetweenUλ(1, 2, 4)¶" & 
                            "TRUE           →=IsBetweenUλ(2, 2, 4)¶" & 
                            "FALSE          →=IsBetweenUλ(2, 2, 4, FALSE)", 
                            "→", "¶"
                            )
                        ),
    //  Check inputs - Set optional arguments defaults
        Inclusive,      IF( ISLOGICAL( Inclusive), Inclusive, TRUE),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Value),
                            ISOMITTED( Low),
                            ISOMITTED( Hi)
                        ),
    //  Procedure
        Result,         IF( Inclusive, 
                            (Value >= Low) * (Value <= Hi) = 1, 
                            (Value > Low) * (Value < Hi) = 1
                        ),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )
);



/*  FUNCTION NAME:      IsInListUλ
    DESCRIPTION:*//**Determine whether a value is one of a list of items*/
IsInListUλ = LAMBDA(
//  Parameter Declarations
    [Value],
    [List],
//  Help
    LET(Help,           TRIM(TEXTSPLIT( 
                            "FUNCTION:      →IsInListUλ( Value, List)¶" &
                            "DESCRIPTION:   →Determine if a value is one of a list of items.¶" & 
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit/blob/main/docs/workbook-use.md¶" &
                            "VERSION:       →24 Oct 2026¶" &
                            "PARAMETERS:    →¶" & 
                            "Value          →(Required) A value or array of values to compare.¶" & 
                            "List           →(Required) A list of values to check: a row, a column or a grid.¶" &
                            "NOTE!          →The comparison is exact and case-insensitive; * and ? are ordinary characters.¶" &
                            "→¶" &
                            "EXAMPLES:      →¶" & 
                            "Result         →Formula¶" &
                            "FALSE          →=IsInListUλ(1, {2,3,4})¶" & 
                            "TRUE           →=IsInListUλ(1, {1,3,5})¶" & 
                            "→", "→", "¶"
                            )
                        ),
    //  Check inputs - Omitted required arguments
        Help?,          OR( ISOMITTED( Value),
                            ISOMITTED( List)
                        ),
    //  Procedure
    //  Flatten the list first: MATCH needs one dimension and treats * and ? as wildcards
        Result,         ISNUMBER( XMATCH( Value, TOCOL( List), 0)),
    //  Return Result
        CHOOSE(Help? + 1, Result, Help)
    )
);



/*  FUNCTION NAME:  CheckRollUpUλ
    DESCRIPTION:*//**Rolls a model's check results up into one status: OK with the number checked, or FAIL with the number failing and the first failing label*/
CheckRollUpUλ = LAMBDA(
//  Parameter Declaration
    [Checks],
    [Labels],
    [Tolerance],
    LET(
    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:      →CheckRollUpUλ(Checks, [Labels], [Tolerance])¶" &
                            "DESCRIPTION:   →Roll check results up into one status for a banner or summary cell.¶NOTES!         →Each cell of Checks is one check result. TRUE passes and FALSE fails. A¶               →number is a difference: it passes when its size is at most Tolerance and¶               →fails above it. An error or any other text fails. Blank cells and empty¶               →text are not checks. Returns one text value: OK and the number of checks¶               →when every check passes, or FAIL, the number failing, the number checked¶               →and the label of the first failure, reading across each row in turn.¶               →Without Labels, or with a blank label, the failure is named by its¶               →position in that order. A range with no check result fails, so an empty¶               →roll-up never reads as OK.¶" &
                            "WEBPAGE:       →https://github.com/ryanduguid/Ozzit/blob/main/docs/workbook-use.md¶" &
                            "VERSION:       →26 Nov 2026¶" &
                            "PARAMETERS:    →¶" &
                            "Checks         →(Required) A row, column or grid of check results.¶" &
                            "Labels         →(Optional) A label for each check, the same shape as Checks.¶" &
                            "Tolerance      →(Optional) The largest difference that passes, 0 or more. Defaults to 0.¶" &
                            "EXAMPLES:      →¶" &
                            "→Formula (oz is assumed to be the module's name)¶" &
                            "→=oz.CheckRollUpUλ({TRUE,0,0.004}, {""Balance"",""Cash"",""GST""}, 0.005)¶" &
                            "→Result¶" &
                            "→OK: 3 checks pass¶" &
                            "→=oz.CheckRollUpUλ({TRUE,FALSE,12.5}, {""Balance"",""Cash"",""GST""})¶" &
                            "→Result¶" &
                            "→FAIL: 2 of 3 checks fail, first: Cash",
                            "→", "¶"
                        )),
    //  Check inputs - Omitted required arguments
        Help?,          ISOMITTED( Checks),
    //  Labels, when given, match Checks cell for cell
        ValidShape?,    IF( ISOMITTED( Labels), TRUE, IFERROR( AND( ROWS( Labels) = ROWS( Checks), COLUMNS( Labels) = COLUMNS( Checks)), FALSE)),
        ShapeMessage,   "CheckRollUpUλ needs Labels the same shape as Checks.",
    //  Tolerance one number, 0 or more; omitted is 0
        Allowed,        IF( ISOMITTED( Tolerance), 0, Tolerance),
        ValidTolerance?, IFERROR( AND( ROWS( Allowed) = 1, COLUMNS( Allowed) = 1, ISNUMBER( Allowed), Allowed >= 0), FALSE),
        ToleranceMessage, "CheckRollUpUλ needs Tolerance to be one number, 0 or more.",
    //  Procedure - read across each row in turn; a blank cell or empty text is not a check
        Values,         TOCOL( Checks),
        Skip,           TOCOL( ISBLANK( Checks)) + IFERROR( Values = "", FALSE) > 0,
    //  TRUE passes, a number passes within Allowed, and FALSE, an error or other text fails
        Pass,           IF( ISLOGICAL( Values), Values, IF( ISNUMBER( Values), ABS( Values) <= Allowed, FALSE)),
        Fail,           NOT( Skip) * NOT( Pass),
        Failed,         SUM( Fail),
        Checked,        SUM( --NOT( Skip)),
        First,          XMATCH( 1, Fail),
    //  The first failure's label, or its position when it has none
        Name,           IFERROR( LET( Label, INDEX( TOCOL( IF( ISBLANK( Labels), "", Labels)), First), IF( Label = "", NA(), Label)), "check " & First),
        Result,         IF( Checked = 0, "FAIL: no check results",
                            IF( Failed = 0, "OK: " & IF( Checked = 1, "1 check passes", Checked & " checks pass"),
                                "FAIL: " & Failed & " of " & Checked & " checks fail, first: " & Name)),
    //  Return Result, help or a message
        Return,         IF( Help?, 2, IF( NOT( ValidShape?), 3, IF( NOT( ValidTolerance?), 4, 1))),
        CHOOSE( Return, Result, Help, ShapeMessage, ToleranceMessage)
    )
);

    //  Help
        Help,           TRIM(TEXTSPLIT(
                            "FUNCTION:          →RangeToDAUλ(Range, [Rows], [Columns])¶" &
                            "DESCRIPTION:       →Convert a range reference to a dynamic array reference.¶" & 
                            "SEE ALSO:      →Excel 365 now has TRIMRANGE and trim references (.:.) for this.¶" &
                            "WEBPAGE:           →https://github.com/ryanduguid/Ozzit/blob/main/docs/workbook-use.md¶" &
                            "VERSION:           →24 Oct 2026¶" &
                            "PARAMETERS:        →¶" & 
                            "Range              →(Required) A reference to the range to be converted.¶" &
                            "Rows               →(Optional) Number of rows for the dynamic array. Defaults to 1.¶" &
                            "Columns            →(Optional) Number of columns for the dynamic array. Defaults to 1.¶" &
                            "→¶" &
                            "EXAMPLES:          →=RangeToDAUλ(A1, , ModelPeriodCount)",
                            "→", "¶"
                            )
                        ),  
    //  Check inputs - Omitted required arguments
        Help?,          ISOMITTED( Range),
    //  Check inputs - Set optional arguments' defaults
        Rows,           IF( ISNUMBER( Rows), Rows, 1),
        Columns,        IF( ISNUMBER( Columns), Columns, 1),
    //  Procedure
        Result,         OFFSET( Range, 0, 0, Rows, Columns), 
    //  Return Result or help
        CHOOSE(Help? + 1, Result, Help)
    )
);
